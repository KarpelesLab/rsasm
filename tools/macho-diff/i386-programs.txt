# Whole programs in a 32-bit Mach-O object.
#
# The structures are narrower throughout -- an `mh_magic` header with no
# reserved word, an `LC_SEGMENT` of `section` rather than `section_64`
# records, `nlist` rather than `nlist_64`, and a file padded to four bytes --
# and the relocations are Mach-O's generic ones, which have a scattered
# record where the 64-bit machines have an addend or a `SUBTRACTOR`. The
# cases below walk through what needs one: a difference of two symbols, which
# is a `SECTDIFF` or a `LOCAL_SECTDIFF` with the subtracted address in the
# `PAIR` behind it, and a reference some way into a symbol, where the record
# carries the address the field was measured from because a section cannot
# say which symbol of it that was.
=== hello, as clang writes it
	.section	__TEXT,__text,regular,pure_instructions
	.globl	_main
	.align	4, 0x90
_main:
	pushl	%ebp
	movl	%esp, %ebp
	subl	$8, %esp
	calll	L0$pb
L0$pb:
	popl	%eax
	leal	L_.str-L0$pb(%eax), %eax
	movl	%eax, (%esp)
	calll	_printf
	xorl	%eax, %eax
	addl	$8, %esp
	popl	%ebp
	retl
	.section	__TEXT,__cstring,cstring_literals
L_.str:
	.asciz	"Hello from rsasm!\n"
	.subsections_via_symbols
