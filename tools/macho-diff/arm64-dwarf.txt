# The line table, in the shapes tools/dwarf-diff checks for an ELF object,
# written in Darwin's assembly. There are no frames here: llvm-mc writes a
# compact unwind word beside `__eh_frame` on arm64, so rsasm refuses `.cfi_*`
# rather than describe a frame half way (see tests/macho.rs).
=== one row
	.file 1 "a.c"
	.text
	.loc 1 1
	nop
=== rows, columns and advances
	.file 1 "a.c"
	.text
_f:
	.loc 1 1 0
	nop
	.loc 1 2 0
	nop
	nop
	.loc 1 3 4
	nop
=== flags and options
	.file 1 "a.c"
	.text
	.loc 1 3 1
	nop
	.loc 1 4 3 is_stmt 0
	nop
	.loc 1 5 7 discriminator 3
	nop
	.loc 1 6 prologue_end
	nop
	.loc 1 7 epilogue_begin
	nop
	.loc 1 8 basic_block
	nop
	.loc 1 9 isa 2
	nop
	.loc 1 10 is_stmt 1
	nop
=== two sequences
	.file 1 "a.c"
	.file 2 "b.h"
	.text
	.loc 1 1
	nop
	.loc 2 20 2
	nop
	.section __TEXT,__other
	.loc 1 30
	nop
	nop
	.text
	.loc 1 2
	nop
=== line zero
	.file 1 "a.c"
	.text
	.loc 1 0
	nop
	.loc 1 3
	nop
	.loc 1 0
	nop
=== large address advances
	.file 1 "a.c"
	.text
	.loc 1 3
	nop
	.loc 1 4
	.space 300
	nop
	.loc 1 5
	.space 70000
	nop
	.loc 1 6
	.space 36
=== data after the last row
	.file 1 "a.c"
	.text
	.loc 1 1
	nop
	.space 40
=== directories in file names
	.file 1 "src/a.c"
	.file 2 "/usr/include/stdio.h"
	.file 3 "src/b.h"
	.text
	.loc 1 1
	nop
	.loc 2 2
	nop
	.loc 3 3
	nop
