# The line table and the frame table of a 32-bit Mach-O object, in the shapes
# tools/dwarf-diff checks for an ELF object.
#
# The debugging sections hold four-byte addresses here, and the relocations
# that reach into them name a section apiece, as a 32-bit object's do
# everywhere. The CIE's data alignment is -4 and its return address column 8,
# as the i386 psABI has them, but its CFA names register 5, since Darwin
# numbers `%esp` and `%ebp` its own way in the unwinding tables and the other
# way in `.debug_frame`; and the FDE's address is the pointer-sized
# PC-relative one llvm-mc writes in a Mach-O object, not the four-byte
# `DW_EH_PE_sdata4` GNU as writes in an ELF one.
=== one row
	.file 1 "a.c"
	.text
	.loc 1 1
	nop
=== rows, columns and advances
	.file 1 "a.c"
	.text
_f:
	.loc 1 1 0
	nop
	.loc 1 2 0
	nop
	nop
	.loc 1 3 4
	nop
=== flags and options
	.file 1 "a.c"
	.text
	.loc 1 3 1
	nop
	.loc 1 4 3 is_stmt 0
	nop
	.loc 1 5 7 discriminator 3
	nop
	.loc 1 6 prologue_end
	nop
	.loc 1 7 epilogue_begin
	nop
	.loc 1 8 basic_block
	nop
	.loc 1 9 isa 2
	nop
	.loc 1 10 is_stmt 1
	nop
=== two sequences
	.file 1 "a.c"
	.file 2 "b.h"
	.text
	.loc 1 1
	nop
	.section __TEXT,__other
	.loc 2 1
	nop
	nop
=== large address advances
	.file 1 "a.c"
	.text
	.loc 1 1
	.space 300
	.loc 1 2
	.space 70000
	.loc 1 3
	nop
=== directories in file names
	.file 1 "/work/dir" "src/a.c"

	.loc 2 2

	.loc 3 3
	nop
=== a file table with no rows
	.file 1 "a.c"
	.text
	nop
=== DWARF 5 with file 0
	.file 0 "/work/dir" "src/a.c" md5 0x00112233445566778899aabbccddeeff
	.file 1 "src/b.c" md5 0x0102030405060708090a0b0c0d0e0f10

	nop
=== a frame
	.text
	.globl _f
_f:
	.cfi_startproc
	nop
	.cfi_def_cfa_offset 8
	nop
	.cfi_offset 3, -8
	nop
	.cfi_restore 3
	nop
	.cfi_endproc
=== a frame pointer prologue
	.text
	.globl _f
_f:
	.cfi_startproc
	pushl %ebp
	.cfi_def_cfa_offset 8
	.cfi_offset %ebp, -8
	movl %esp, %ebp
	.cfi_def_cfa_register %ebp
	popl %ebp
	retl
	.cfi_endproc
=== every instruction
	.text
	.globl _f
_f:
	.cfi_startproc
	nop
	.cfi_def_cfa_offset 24
	.cfi_offset 3, -24
	.cfi_offset 4, 16
	.space 100
	.cfi_rel_offset 6, 8
	.cfi_adjust_cfa_offset 8
	.cfi_def_cfa 5, 16
	.cfi_def_cfa 6, -8
	.space 300
	.cfi_remember_state
	.cfi_def_cfa_register 4
	.cfi_register 3, 4
	.cfi_same_value 2
	.cfi_restore 3
	.cfi_undefined 9
	.space 70000
	.cfi_restore_state
	.cfi_adjust_cfa_offset -8
	.cfi_val_offset 3, -16
	.cfi_escape 0x2e, 0x10
	.cfi_def_cfa_offset -16
	nop
	.cfi_endproc

=== lines and frames together
	.file 1 "a.c"
	.text
	.globl _f
_f:
	.loc 1 3 0
	.cfi_startproc
	pushl %ebp
	.cfi_def_cfa_offset 8
	.cfi_offset %ebp, -8
	movl %esp, %ebp
	.cfi_def_cfa_register %ebp
	.loc 1 4 5 prologue_end
	nop
	.loc 1 5 1 epilogue_begin
	popl %ebp
	retl
	.cfi_endproc
