# i386's compact unwind table, which a deployment target of macOS 10.6 or
# later asks for as it does on x86-64.
#
# The word is the same algorithm with narrower units: a four-byte stack slot,
# so the adjustment reaches 1,020 bytes rather than 2,040 and a frame
# pointer's saved registers sit twelve bytes below the CFA, and the six
# registers the word can name are the ones the 32-bit ABI preserves. The
# entry is twenty bytes rather than thirty-two.
#
# A CIE cannot name a personality routine at all here: Darwin's CIE reaches
# one through the GOT, a 32-bit object has no relocation for a GOT entry, and
# llvm-mc leaves the distance to the routine itself -- which it writes as a
# difference of two symbols and so refuses for a routine the object does not
# define, which is every routine a compiler names. Both assemblers refuse it.
=== a frame pointer prologue
	.text
	.globl _f
_f:
	.cfi_startproc
	pushl	%ebp
	.cfi_def_cfa_offset 8
	.cfi_offset %ebp, -8
	movl	%esp, %ebp
	.cfi_def_cfa_register %ebp
	popl	%ebp
	retl
	.cfi_endproc
=== a frame pointer prologue that saves one register
	.text
	.globl _f
_f:
	.cfi_startproc
	.cfi_def_cfa_offset 8
	.cfi_offset %ebp, -8
	.cfi_def_cfa_register %ebp
	.cfi_offset %ebx, -12
	retl
	.cfi_endproc
=== a frame pointer prologue that saves five

_f:

	.cfi_offset %ecx, -12
	.cfi_offset %edx, -16
	.cfi_offset %edi, -20
	.cfi_offset %esi, -24
	.cfi_offset %ebp, -28
	retl
	.cfi_endproc

=== a language-specific data area
	.text

_f:

	.section __TEXT,__gcc_except_tab
Lexception0:
	.byte 0xff
=== a whole function as clang writes it
	.section	__TEXT,__text,regular,pure_instructions
	.globl	_main
	.align	4, 0x90
_main:
	.cfi_startproc
	pushl	%ebp
	.cfi_def_cfa_offset 8
	.cfi_offset %ebp, -8
	movl	%esp, %ebp
	.cfi_def_cfa_register %ebp
	subl	$8, %esp
	movl	$0, -4(%ebp)
	addl	$8, %esp
	popl	%ebp
	retl
	.cfi_endproc
	.subsections_via_symbols
