# The frame table of an object whose deployment target is a macOS that reads
# a compact unwind table -- 10.6, the first of them.
#
# A frame on x86-64 is described twice over: as a compact unwind word in
# `__LD,__compact_unwind`, which the linker reads first, and in
# `__TEXT,__eh_frame`, which it reads where the word tells it to and which is
# written for every frame either way, since x86-64's word alone does not
# describe one. The cases walk the shapes the word has -- a frame pointer
# prologue, a frameless adjustment the word holds, a frameless one the word
# only points at -- the callee-saved registers each form holds, and the
# shapes that fall back to the frame table.
=== a frame pointer prologue
	.text
	.globl _f
_f:
	.cfi_startproc
	pushq	%rbp
	.cfi_def_cfa_offset 16
	.cfi_offset %rbp, -16
	movq	%rsp, %rbp
	.cfi_def_cfa_register %rbp
	popq	%rbp
	retq
	.cfi_endproc

=== a whole function as clang writes it
	.section	__TEXT,__text,regular,pure_instructions
	.globl	_main
	.p2align	4, 0x90
_main:
	.cfi_startproc
	pushq	%rbp
	.cfi_def_cfa_offset 16
	.cfi_offset %rbp, -16
	movq	%rsp, %rbp
	.cfi_def_cfa_register %rbp
	subq	$16, %rsp
	movl	$0, -4(%rbp)
	leaq	L_.str(%rip), %rdi
	callq	_printf
	xorl	%eax, %eax
	addq	$16, %rsp
	popq	%rbp
	retq
	.cfi_endproc
	.section	__TEXT,__cstring,cstring_literals
L_.str:
	.asciz	"hello\n"
	.subsections_via_symbols
