=== references to symbols in their own section, by binding
        .text
        .globl  glob, hid, prot, intl, galias, func
        .hidden hid
        .protected prot
        .internal intl
        .weak   weak1, walias, whid
        .hidden whid
        .type   func, %function
        .hidden lhid
start:
        call loc
        call glob
        call weak1
        call hid
        call prot
        call intl
        call whid
        call lhid
        call func
        call late
        call lweak
        call lalias
        call lwalias
        call galias
        call walias
        jmp loc
        jmp glob
        jmp weak1
        jmp lalias
        jmp walias
        jz loc
        jz glob
        jz weak1
        jz lalias
        jz walias
        lea loc(%rip), %rax
        lea glob(%rip), %rax
        lea weak1(%rip), %rax
        lea lalias(%rip), %rax
        lea galias(%rip), %rax
        call loc@PLT
        call glob@PLT
        call weak1@PLT
        call lalias@PLT
        call galias@PLT
        jmp loc@PLT
        jmp glob@PLT
        jmp weak1@PLT
        jmp lalias@PLT
        jmp galias@PLT
        nop

glob:   nop
weak1:  nop
hid:    nop
prot:   nop

whid:   nop
lhid:   nop
func:   nop
late:   nop

        .globl  late
        .weak   lweak
        .set    lalias, glob
        .set    lwalias, weak1
        .set    galias, loc
        .set    walias, loc

# GNU as writes a name only `.type`, `.size`, `.globl`, `.local` or a
# visibility mentions as a global undefined symbol, and leaves out one
# `.weak` names, which stays weak after `.globl`. llvm-mc keeps the weak
# one, leaves the sized one out, makes the `.local` one local, and
# refuses `.globl` after `.weak`.

        .internal intl
