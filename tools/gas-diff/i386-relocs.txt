# GNU as writes a name only `.type`, `.size`, `.globl`, `.local` or a
# visibility mentions as a global undefined symbol, and leaves out one
# `.weak` names, which stays weak after `.globl`. llvm-mc keeps the weak
# one, leaves the sized one out, makes the `.local` one local, and
# refuses `.globl` after `.weak`.

# reference to the symbol: GNU as says `junk `@FOOBAR' after expression`.
=== refused: a modifier the backend does not know, in an operand

# `@SIZE` is the size of the symbol rather than its address, and GNU as
# settles it here for a symbol the file defines and does not export, since
# nothing can change it after that: the number goes straight into the field
# with no relocation at all. A `.size` given after the reference still
# counts, a `.set` alias carries the size of the symbol it names without the
# offset it adds -- an alias of a symbol not yet defined carries only the
# plain equate's, since an offset is what settles the equate into a number --
# and a weak definition is settled like a local one: the test is
# `STB_GLOBAL`, not what a linker may preempt. llvm-mc 22 has no row
# for any of this: it crashes on `@SIZE` in an i386 object, whatever the
# symbol.
=== @SIZE of a symbol this file settles
        .set    fwd, late
        .set    fwdoff, late + 1
        .data
loc:    .space  24
        .size   loc, 24
fn:     .space  7
        .type   fn, %function
        .size   fn, . - fn
        .weak   weaksz
weaksz: .space  9
        .size   weaksz, 9
late:   .space  3
        .lcomm  lc, 17
        .local  lcm
        .comm   lcm, 12, 4
        .set    alias, loc
        .set    aliasoff, loc + 4
        .text
        .long   loc@SIZE, fn@SIZE, weaksz@SIZE, late@SIZE
        .long   lc@SIZE, lcm@SIZE, alias@SIZE, aliasoff@SIZE
        .long   fwd@SIZE, fwdoff@SIZE
        .long   loc@SIZE + 8, loc@SIZE - 1
        movl    $loc@SIZE, %eax
        movl    loc@SIZE(%eax), %ecx
        pushl   $loc@SIZE
        .size   late, 9

# The other half, which the linker settles: a symbol this file exports may
# be replaced by a definition of another size, a common block is one such,
# and an undefined symbol has no size here at all. i386 is REL, so the
# addend stays in the field where x86-64 puts it in the relocation entry.
# Every four-byte field takes the modifier here, a displacement included:
# nothing in 32-bit mode sign-extends one.
=== @SIZE the linker settles
        .globl  glob, gfunc
        .comm   cm, 40, 8
        .weak   wund
        .text
        .long   und@SIZE, und@SIZE + 8, und@SIZE - 4
        .long   glob@SIZE, gfunc@SIZE, cm@SIZE, wund@SIZE
        movl    $und@SIZE, %eax
        movl    und@SIZE, %ecx
        movl    und@SIZE(%eax), %ecx
        leal    und@SIZE(%eax), %ebx
        addl    $und@SIZE, %eax
        pushl   $und@SIZE
        .data
glob:   .space  16
        .size   glob, 16
        .type   gfunc, %function
gfunc:  .space  4
        .size   gfunc, 4

# i386 has `R_386_SIZE32` and nothing wider or narrower, so only a four-byte
# field takes the modifier, whether or not the size was there to write. A
# size is also something the file states about a symbol, so `@SIZE` needs
# one: GNU as calls it an invalid expression otherwise.
=== refused: @SIZE in a two-byte field
        .data
        .word   und@SIZE

=== refused: @SIZE in an eight-byte field
        .data
        .quad   und@SIZE

=== refused: @SIZE of a symbol set to a number
        .set    n, 7
        .data
        .long   n@SIZE
