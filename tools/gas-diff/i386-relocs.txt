# i386 objects, compared with GNU as's on the .text bytes and on every
# relocation (offset, type and symbol). i386 relocations are REL, so the
# addend is in the bytes.

=== calls and jumps to an undefined symbol
    call ext
    jmp ext
    je ext
    call ext@PLT
    jmp ext@PLT

=== absolute references at every width
    movl $ext, %eax
    movw $ext, %ax
    movb $ext, %al
    movl ext, %eax
    movl ext+4(%ebx), %ecx
    pushl $ext
    addl $ext, %edx
    int $ext
    .long ext
    .word ext
    .byte ext
    .long ext - .

=== far pointers to symbols
    ljmp $seg, $ext
    lcall $0x10, $ext+8

=== GOT loads GNU as marks relaxable
    movl ext@GOT(%ebx), %eax
    cmpl ext@GOT(%ebx), %ecx
    addl ext@GOT(%ebx), %edx
    testl %esi, ext@GOT(%ebx)
    call *ext@GOT(%ebx)
    jmp *ext@GOT(%ebx)
    pushl ext@GOT(%ebx)
    movl ext@GOT, %eax
    movl ext@GOT+4(%ebx), %eax

=== GOT references the linker cannot relax
    movl %eax, ext@GOT(%ebx)
    leal ext@GOT(%ebx), %eax
    popl ext@GOT(%ebx)
    movzbl ext@GOT(%ebx), %eax
    movl ext@GOT(,%ebx,1), %eax
    movl $ext@GOT, %eax
    .long ext@GOT

=== the GOT itself, and offsets from it
    call 1f
1:  popl %ebx
    addl $_GLOBAL_OFFSET_TABLE_+(.-1b), %ebx
    leal _GLOBAL_OFFSET_TABLE_(%ebx), %eax
    movl ext@GOTOFF(%ebx), %eax
    leal ext@GOTOFF+4(%ebx), %ecx
    movl $ext@GOTOFF, %eax
    .long ext@GOTOFF

=== thread-local storage models
    leal ext@TLSGD(,%ebx,1), %eax
    leal ext@TLSLDM(%ebx), %eax
    leal ext@DTPOFF(%eax), %eax
    movl %gs:ext@NTPOFF, %eax
    movl $ext@TPOFF, %eax
    movl ext@INDNTPOFF, %eax
    movl ext@GOTNTPOFF(%ebx), %eax
    movl ext@GOTTPOFF(%ebx), %eax
    .long ext@DTPOFF

=== 16-bit code: word relocations
    .code16
    call ext
    jmp ext
    jne ext
    movw $ext, %ax
    movw ext, %bx
    movw ext+2(%bx,%si), %cx
    ljmp $0, $ext
    pushw $ext

=== 16-bit code: byte relocations
    .code16
    jcxz ext
    loop ext
    movb $ext, %al

=== 16-bit code: 32-bit operands and addresses keep 32-bit relocations
    .code16
    movl $ext, %eax
    movl ext(%eax), %ebx
    calll ext
    ljmpl $0x10, $ext

=== code16gcc calls are 32-bit
    .code16gcc
    call ext
    jmp ext

=== local symbols are relocated against their section
    movl $local, %eax
    call local
    movl local+2, %ecx
    .data
local:
    .long 0
