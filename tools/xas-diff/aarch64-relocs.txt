# AArch64 against GNU as, as whole objects: the literal pools, which are GNU
# as's feature and llvm-mc's differently (see run.sh), with their mapping
# symbols and the relocations a symbol in a pool needs.

=== a pool of every width, written at the end of the section
	.text
	.globl	f
f:
	ldr	w0, =0x12345678
	ldr	x1, =0x123456789
	ldr	s2, =0x3fc00000
	ldr	d3, =0x3ff8000000000000
	ldr	q4, =0x1122334455667788
	ldrsw	x5, =-1
	ret

=== the same value twice is one entry, and each width has its own run
	.text
f:
	ldr	x0, =0x123456789
	ldr	x1, =0x123456789
	ldr	w2, =7
	ldr	x3, =7
	ldr	w4, =7
	ret

=== .ltorg writes the pool out where it stands, and starts another
	.text
f:
	ldr	x0, =0x1111222233334444
	ldr	w1, =0x55556666
	ret
	.ltorg
g:
	ldr	x2, =0x7777888899990000
	ret

=== .pool is .ltorg
	.text
f:
	ldr	x0, =0xdeadbeefcafe
	ret
	.pool
g:
	ret

=== a symbol in a pool is relocated, with its addend
	.text
	.globl	f
f:
	ldr	x0, =far
	ldr	w1, =far
	ldr	x2, =far + 16
	ldr	x3, =local
	ldr	x4, =far
local:
	ret
	.globl	far

=== each section has its own pool
	.text
f:
	ldr	x0, =0x1234567890
	ret
	.section	.text.other,"ax",@progbits
g:
	ldr	x1, =0x9876543210
	ret

=== a pool after data in a code section
	.text
f:
	ldr	x0, =0x5555aaaa5555aaaa
	ret
	.byte	1, 2, 3
	.ltorg

=== the pool is aligned and the padding between its runs is code
	.text
f:
	ldr	w0, =1
	ldr	w1, =2
	ldr	w2, =3
	ldr	x3, =0x100000000
	ret
	.ltorg

=== ldr of a pool entry reaches a megabyte away
	.text
f:
	ldr	x0, =0x123456789
	.space	1024 * 1024 - 16
	ret
	.ltorg

=== refused: an entry the load cannot reach
	.text
f:
	ldr	x0, =0x123456789
	.space	1024 * 1024
	ret
	.ltorg

=== refused: only ldr loads from a literal pool
	.text
f:
	str	x0, =0x123456789

=== refused: a literal pool value is not an address operand
	.text
f:
	add	x0, x1, =4

# ---- thread-local storage ----
#
# Every access model's operators, in each instruction GNU as takes them on.
# None is ever resolved here: the linker lays out the thread-local block and
# may rewrite a whole sequence into another model, so even a variable defined
# in this file is relocated, by name, and an undefined one becomes `STT_TLS`.
# llvm-mc 22 knows neither `:tlsgd:`, `:tlsldm:` and their `_lo12`, `_g1`
# and `_g0_nc` forms, nor `:tlsdesc_off_g1:`, `.tlsdescadd`, `.tlsdescldr`
# or `%dtprel`, so those are here and not in tools/mc-diff; the rows where
# the two references write something different say so.

=== local exec: the thread pointer offset, sixteen and twelve bits at a time
# `movz` stays `movz` for a signed group, where llvm-mc writes `movn`: the
# linker chooses between the two once it knows the sign. `:tprel_hi12:`
# shifts its `add` by twelve whether or not `lsl #12` is written, and after
# any `lsl` GNU as can encode; `:tprel:` is `:tprel_lo12:` on an `add`.
	.text
	.globl	f
f:
	movz	x0, :tprel_g2:v
	movk	x0, :tprel_g1_nc:v
	movk	x0, :tprel_g0_nc:v
	movz	x1, :tprel_g1:v
	movn	x2, :tprel_g0:v+8
	movz	w3, :tprel_g1:v
	movk	w3, :tprel_g0_nc:v
	mrs	x4, tpidr_el0
	add	x4, x4, :tprel_hi12:v
	add	x4, x4, :tprel_lo12_nc:v
	add	x5, x4, :tprel_hi12:v, lsl #12
	add	x5, x4, :tprel_hi12:v, lsl #0
	add	x5, x4, :tprel_hi12:v, lsl #3
	add	x5, x4, :tprel_lo12:v, lsl #12
	add	w6, w6, :tprel_lo12:v
	add	x7, x7, :tprel:v
	add	sp, x4, :tprel_lo12:v
	ret

=== local exec: loads and stores, one relocation per access size
	.text
f:
	ldrb	w0, [x1, :tprel_lo12:v]
	ldrsb	x0, [x1, :tprel_lo12_nc:v]
	ldrh	w0, [x1, :tprel_lo12:v]
	strh	w0, [x1, :tprel_lo12_nc:v]
	ldr	w0, [x1, :tprel_lo12:v]
	ldrsw	x0, [x1, :tprel_lo12_nc:v]
	ldr	x0, [x1, #:tprel_lo12:v+16]
	str	x0, [x1, :tprel_lo12_nc:v]
	ldr	b0, [x1, :tprel_lo12:v]
	ldr	h0, [x1, :tprel_lo12:v]
	ldr	s0, [x1, :tprel_lo12:v]
	ldr	d0, [x1, :tprel_lo12_nc:v]
	prfm	pldl1keep, [x1, :tprel_lo12:v]

=== local dynamic: the module's block, and the offset within it
# `:dtprel_hi12:` takes the shift only as written, unlike `:tprel_hi12:`;
# llvm-mc shifts it either way. `movk` takes every `:dtprel_gN:`, which it
# does not take `:tprel_g1:` or `:tprel_g0:`.
	.text
f:
	adrp	x0, :tlsldm:v
	add	x0, x0, :tlsldm_lo12_nc:v
	adr	x1, :tlsldm:v
	bl	__tls_get_addr
	movz	x1, :dtprel_g2:v
	movk	x1, :dtprel_g1:v
	movk	x1, :dtprel_g0:v
	movz	x2, :dtprel_g1_nc:v
	movk	x2, :dtprel_g0_nc:v
	add	x0, x0, :dtprel_hi12:v
	add	x0, x0, :dtprel_hi12:v, lsl #12
	add	x0, x0, :dtprel_lo12_nc:v
	add	x3, x3, :dtprel_lo12:v
	ldrb	w4, [x0, :dtprel_lo12:v]
	ldrh	w4, [x0, :dtprel_lo12_nc:v]
	ldr	w4, [x0, :dtprel_lo12:v]
	ldr	x4, [x0, :dtprel_lo12_nc:v]

=== general dynamic, and the move-wide forms of the GOT-based models
# `:tlsgd_g1:` is refused on `movk`, `:tlsdesc_off_g1:` and
# `:gottprel_g1:` are not.
	.text
f:
	adrp	x0, :tlsgd:v
	add	x0, x0, :tlsgd_lo12:v
	adr	x1, :tlsgd:v+4
	movz	x2, :tlsgd_g1:v
	movk	x2, :tlsgd_g0_nc:v
	movz	x3, :gottprel_g1:v
	movk	x3, :gottprel_g0_nc:v
	movk	x4, :tlsdesc_off_g1:v
	movk	x4, :tlsdesc_off_g0_nc:v
	movz	w5, :tlsdesc_off_g1:v

=== initial exec: the offset from a GOT slot
# `:gottprel_lo12:` gets the 64-bit slot's relocation on any load, a byte or
# 128-bit one included, as `:tlsdesc_lo12:` does; llvm-mc refuses those.
	.text
f:
	adrp	x0, :gottprel:v
	ldr	x0, [x0, :gottprel_lo12:v]
	ldr	w1, [x0, :gottprel_lo12:v]
	ldrb	w1, [x0, :gottprel_lo12:v]
	ldr	q1, [x0, :gottprel_lo12:v]
	ldr	x2, :gottprel:v
	ldr	d2, :gottprel:v
	mrs	x3, tpidr_el0
	add	x0, x0, x3

=== TLS descriptors, and the marks on the sequence's instructions
# `.tlsdesccall`, `.tlsdescadd` and `.tlsdescldr` put a relocation covering
# no bytes on whatever comes next, and name what they are given, with an
# addend, without making it thread-local. A number is taken and nothing
# written for it.
	.text
f:
	adrp	x0, :tlsdesc:v
	.tlsdescldr	v
	ldr	x1, [x0, :tlsdesc_lo12:v]
	.tlsdescadd	v
	add	x0, x0, :tlsdesc_lo12:v
	.tlsdesccall	v
	blr	x1
	adr	x0, :tlsdesc:w
	ldr	x1, :tlsdesc:w
	ldr	w1, [x0, :tlsdesc_lo12:w]
	.tlsdesccall	u+4
	.tlsdesccall	u
lab:	blr	x1
	.tlsdesccall	2
	nop
	.tlsdesccall	loc
	nop
loc:	.tlsdescadd	loc
	.word	0

=== variables defined here, by binding
	.section	.tbss,"awT",@nobits
	.p2align	3
tb:	.space	8
	.globl	tg
tg:	.space	8
	.section	.tdata,"awT",@progbits
td:	.quad	1
	.tls_common	tc, 8, 8
	.text
f:
	add	x0, x0, :tprel_lo12:tb
	add	x0, x0, :tprel_lo12:tg+4
	movz	x1, :tprel_g1:td
	adrp	x2, :gottprel:tg
	ldr	x2, [x2, :gottprel_lo12:tg]
	adrp	x3, :tlsdesc:tc
	.data
	.xword	%dtprel(tb)

=== the type a thread-local access gives an undefined symbol
# An `object` becomes `STT_TLS` as a plain undefined symbol does.
	.type	o, %object
	.text
	add	x0, x0, :tprel_lo12:o
	add	x0, x0, :tprel_lo12:later
	.type	later, %object

=== %dtprel, in the two eight-byte directives GNU as reads it in
	.data
	.xword	%dtprel(v)
	.dword	%dtprel(v+8)
	.xword	1, %dtprel (w), 2

=== a number under an operator on a PC-relative field is an offset
# GNU as folds it before it looks at the operator, and writes what the
# instruction without one would.
	.text
f:
	nop
	adr	x0, :tlsgd:8
	ldr	x1, :gottprel:-4
	cbz	x2, :tlsdesc:8

=== the literal forms of `:tlsdesc:` and `:gottprel:` on compare-and-branch
# GNU as reads a `cbz`, `cbnz`, `tbz` or `tbnz` target with the parser it
# reads a literal load's with, and writes the load's relocation; llvm-mc
# ignores the operator and writes the branch's own.
	.text
f:
	cbz	x0, :tlsdesc:v
	cbnz	w1, :gottprel:v
	tbz	x2, #1, :tlsdesc:v
	tbnz	w3, #5, :gottprel:v

# reserves by its size, and a `.comm` of a `.local` symbol only as asked;
# llvm-mc packs both. A `.local` after the `.comm` changes nothing.
=== objects .lcomm and a .local .comm reserve in .bss

        .lcomm  l4, 4
        .lcomm  l3, 3
        .lcomm  l17, 17
        .lcomm  l2, 2

        .comm   b9, 9

        .comm   late, 4
        .local  late
        .data

# GNU as writes a name only `.type`, `.size`, `.globl`, `.local` or a
# visibility mentions as a global undefined symbol, and leaves out one
# `.weak` names, which stays weak after `.globl`. llvm-mc keeps the weak
# one, leaves the sized one out, makes the `.local` one local, and
# refuses `.globl` after `.weak`.
=== symbols only a directive names
        .type   tfunc, %function
        .type   tobj, %object
        .size   sized, 4
        .hidden hid
        .protected prot
        .internal intl
        .globl  gonly
        .weak   wonly
        .local  lonly
        .weak   wglob
        .globl  wglob
        .weak   wtyped
        .type   wtyped, %function
        .weak   wref, wdef
        .globl  wref, wdef
        .type   .Ltyped, %function
        .hidden .Lhid
        .data
wdef:   .xword  wref
