# AArch64 against GNU as, as whole objects: the literal pools, which are GNU
# as's feature and llvm-mc's differently (see run.sh), with their mapping
# symbols and the relocations a symbol in a pool needs.

=== a pool of every width, written at the end of the section
	.text
	.globl	f
f:
	ldr	w0, =0x12345678
	ldr	x1, =0x123456789
	ldr	s2, =0x3fc00000
	ldr	d3, =0x3ff8000000000000
	ldr	q4, =0x1122334455667788
	ldrsw	x5, =-1
	ret

=== the same value twice is one entry, and each width has its own run
	.text
f:
	ldr	x0, =0x123456789
	ldr	x1, =0x123456789
	ldr	w2, =7
	ldr	x3, =7
	ldr	w4, =7
	ret

=== .ltorg writes the pool out where it stands, and starts another
	.text
f:
	ldr	x0, =0x1111222233334444
	ldr	w1, =0x55556666
	ret
	.ltorg
g:
	ldr	x2, =0x7777888899990000
	ret

=== .pool is .ltorg
	.text
f:
	ldr	x0, =0xdeadbeefcafe
	ret
	.pool
g:
	ret

=== a symbol in a pool is relocated, with its addend
	.text
	.globl	f
f:
	ldr	x0, =far
	ldr	w1, =far
	ldr	x2, =far + 16
	ldr	x3, =local
	ldr	x4, =far
local:
	ret
	.globl	far

=== each section has its own pool
	.text
f:
	ldr	x0, =0x1234567890
	ret
	.section	.text.other,"ax",@progbits
g:
	ldr	x1, =0x9876543210
	ret

=== a pool after data in a code section
	.text
f:
	ldr	x0, =0x5555aaaa5555aaaa
	ret
	.byte	1, 2, 3
	.ltorg

=== the pool is aligned and the padding between its runs is code
	.text
f:
	ldr	w0, =1
	ldr	w1, =2
	ldr	w2, =3
	ldr	x3, =0x100000000
	ret
	.ltorg

=== ldr of a pool entry reaches a megabyte away
	.text
f:
	ldr	x0, =0x123456789
	.space	1024 * 1024 - 16
	ret
	.ltorg

=== refused: an entry the load cannot reach
	.text
f:
	ldr	x0, =0x123456789
	.space	1024 * 1024
	ret
	.ltorg

=== refused: only ldr loads from a literal pool
	.text
f:
	str	x0, =0x123456789

=== refused: a literal pool value is not an address operand
	.text
f:
	add	x0, x1, =4

# GNU as aligns a common block that names no alignment, or 0, to its size
# rounded up to a power of two, at most 16, on every ELF target; llvm-mc
# aligns it to 1.
=== common blocks that name no alignment, by size
        .comm   c0, 0
        .comm   c1, 1
        .comm   c2, 2
        .comm   c3, 3
        .comm   c5, 5
        .comm   c8, 8
        .comm   c9, 9
        .comm   c16, 16
        .comm   c17, 17
        .comm   c100, 100
        .comm   czero, 12, 0
        .comm   codd, 8, 3
        .tls_common tc3, 3
        .tls_common tc40, 40

# Both references reserve these in `.bss`. GNU as aligns what `.lcomm`
# reserves by its size, and a `.comm` of a `.local` symbol only as asked;
# llvm-mc packs both. A `.local` after the `.comm` changes nothing.
=== objects .lcomm and a .local .comm reserve in .bss
        .lcomm  l1, 1
        .lcomm  l4, 4
        .lcomm  l3, 3
        .lcomm  l17, 17
        .lcomm  l2, 2
        .lcomm  l0, 0
        .lcomm  l5, 5
        .local  b1, b9, b2
        .comm   b1, 1
        .comm   b9, 9
        .comm   b2, 2, 32
        .comm   late, 4
        .local  late
        .data
        .xword  l1, l4, l3, l17, l2, l0, l5, b1, b9, b2, late

# GNU as writes a name only `.type`, `.size`, `.globl`, `.local` or a
# visibility mentions as a global undefined symbol, and leaves out one
# `.weak` names, which stays weak after `.globl`. llvm-mc keeps the weak
# one, leaves the sized one out, makes the `.local` one local, and
# refuses `.globl` after `.weak`.
=== symbols only a directive names
        .type   tfunc, %function
        .type   tobj, %object
        .size   sized, 4
        .hidden hid
        .protected prot
        .internal intl
        .globl  gonly
        .weak   wonly
        .local  lonly
        .weak   wglob
        .globl  wglob
        .weak   wtyped
        .type   wtyped, %function
        .weak   wref, wdef
        .globl  wref, wdef
        .type   .Ltyped, %function
        .hidden .Lhid
        .data
wdef:   .xword  wref
