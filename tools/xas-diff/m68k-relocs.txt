=== references to symbols in their own section, by binding
        .text
        .globl  glob, hid, prot, intl, galias, func
        .hidden hid
        .protected prot
        .internal intl
        .weak   weak1, walias, whid
        .hidden whid
        .type   func, %function
        .hidden lhid
start:
        bsr loc
        bsr glob
        bsr weak1
        bsr hid
        bsr prot
        bsr intl
        bsr whid
        bsr lhid
        bsr func
        bsr late
        bsr lweak
        bsr lalias
        bsr lwalias
        bsr galias
        bsr walias
        jbsr loc
        jbsr glob
        jbsr weak1
        jbsr lalias
        jbsr walias
        jra loc
        jra glob
        jra weak1
        jra lalias
        jra walias
        beq loc
        beq glob
        beq weak1
        beq lalias
        beq walias
        lea loc(%pc), %a0
        lea glob(%pc), %a0
        lea weak1(%pc), %a0
        lea lalias(%pc), %a0
        lea galias(%pc), %a0
        nop
loc:    nop
glob:   nop
weak1:  nop
hid:    nop
prot:   nop
intl:   nop
whid:   nop
lhid:   nop
func:   nop
late:   nop
lweak:  nop
        .globl  late
        .weak   lweak
        .set    lalias, glob
        .set    lwalias, weak1
        .set    galias, loc
        .set    walias, loc

=== references back to symbols in their own section
        .text
        .globl  glob
        .weak   weak1
loc:    nop
glob:   nop
weak1:  nop
        .set    lalias, glob
        bsr loc
        bsr glob
        bsr weak1
        bsr lalias
        jbsr loc
        jbsr glob
        jbsr weak1
        jra loc
        jra glob
        jra weak1
        beq loc
        beq glob
        beq weak1

=== references out of their section, by binding
        .text
        .globl  oglob
        .weak   oweak, uweak
        .set    oalias, oloc
        bsr oloc
        bsr oglob
        bsr oweak
        bsr oalias
        bsr ext
        bsr uweak
        nop
        .section .text.other,"ax"
oloc:   nop
oglob:  nop
oweak:  nop

=== data referring to symbols, by binding
        .data
        .globl  glob, galias, hid
        .weak   weak1, walias
        .weak   uweak
        .hidden hid
        .set    lalias, glob
        .set    lwalias, weak1
        .set    galias, loc
        .set    walias, loc
base:   .long   loc, glob, weak1, hid, lalias, lwalias, galias, walias, ext
        .long   loc - ., glob - ., weak1 - ., lalias - ., walias - .
        .long   glob - base, weak1 - base, base - weak1, lalias - base, walias - base
        .long   ext - ., uweak - .
loc:    .long   0
glob:   .long   0
weak1:  .long   0
hid:    .long   0
