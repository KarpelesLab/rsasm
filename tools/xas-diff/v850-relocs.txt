=== references to symbols in their own section, by binding
        .text
        .globl  glob, hid, prot, intl, galias, func
        .hidden hid
        .protected prot
        .internal intl
        .weak   weak1, walias, whid
        .hidden whid
        .type   func, %function
        .hidden lhid
start:
        jarl loc, lp
        jarl glob, lp
        jarl weak1, lp
        jarl hid, lp
        jarl prot, lp
        jarl intl, lp
        jarl whid, lp
        jarl lhid, lp
        jarl func, lp
        jarl late, lp
        jarl lweak, lp
        jarl lalias, lp
        jarl lwalias, lp
        jarl galias, lp
        jarl walias, lp
        jr loc
        jr glob
        jr weak1
        jr lalias
        jr walias
        nop
loc:    nop
glob:   nop
weak1:  nop
hid:    nop
prot:   nop
intl:   nop
whid:   nop
lhid:   nop
func:   nop
late:   nop
lweak:  nop
        .globl  late
        .weak   lweak
        .set    lalias, glob
        .set    lwalias, weak1
        .set    galias, loc
        .set    walias, loc

=== references back to symbols in their own section
        .text
        .globl  glob
        .weak   weak1
loc:    nop
glob:   nop
weak1:  nop
        .set    lalias, glob
        jarl loc, lp
        jarl glob, lp
        jarl weak1, lp
        jarl lalias, lp
        jr loc
        jr glob
        jr weak1

=== references out of their section, by binding
        .text
        .globl  oglob
        .weak   oweak, uweak
        .set    oalias, oloc
        jarl oloc, lp
        jarl oglob, lp
        jarl oweak, lp
        jarl oalias, lp
        jarl ext, lp
        jarl uweak, lp
        nop
        .section .text.other,"ax"
oloc:   nop
oglob:  nop
oweak:  nop

=== data referring to symbols, by binding
        .data
        .globl  glob, galias, hid
        .weak   weak1, walias
        .hidden hid
        .set    lalias, glob
        .set    lwalias, weak1
        .set    galias, loc
        .set    walias, loc
base:   .long   loc, glob, weak1, hid, lalias, lwalias, galias, walias, ext
        .long   loc - ., glob - ., weak1 - ., lalias - ., walias - .
        .long   glob - base, weak1 - base, base - weak1, lalias - base, walias - base
loc:    .long   0
glob:   .long   0
weak1:  .long   0
hid:    .long   0

# GNU as aligns a common block that names no alignment, or 0, to its size
# rounded up to a power of two, at most 16, on every ELF target; llvm-mc
# aligns it to 1.
=== common blocks that name no alignment, by size
        .comm   c0, 0
        .comm   c1, 1
        .comm   c2, 2
        .comm   c3, 3
        .comm   c5, 5
        .comm   c8, 8
        .comm   c9, 9
        .comm   c16, 16
        .comm   c17, 17
        .comm   c100, 100
        .comm   czero, 12, 0
        .comm   codd, 8, 3
        .tls_common tc3, 3
        .tls_common tc40, 40

# Both references reserve these in `.bss`. GNU as aligns what `.lcomm`
# reserves by its size, and a `.comm` of a `.local` symbol only as asked;
# llvm-mc packs both. A `.local` after the `.comm` changes nothing.
=== objects .lcomm and a .local .comm reserve in .bss
        .lcomm  l1, 1
        .lcomm  l4, 4
        .lcomm  l3, 3
        .lcomm  l17, 17
        .lcomm  l2, 2
        .lcomm  l0, 0
        .lcomm  l5, 5
        .local  b1, b9, b2
        .comm   b1, 1
        .comm   b9, 9
        .comm   b2, 2, 32
        .comm   late, 4
        .local  late
        .data
        .long   l1, l4, l3, l17, l2, l0, l5, b1, b9, b2, late

# GNU as writes a name only `.type`, `.size`, `.globl`, `.local` or a
# visibility mentions as a global undefined symbol, and leaves out one
# `.weak` names, which stays weak after `.globl`. llvm-mc keeps the weak
# one, leaves the sized one out, makes the `.local` one local, and
# refuses `.globl` after `.weak`.
=== symbols only a directive names
        .type   tfunc, %function
        .type   tobj, %object
        .size   sized, 4
        .hidden hid
        .protected prot
        .internal intl
        .globl  gonly
        .weak   wonly
        .local  lonly
        .weak   wglob
        .globl  wglob
        .weak   wtyped
        .type   wtyped, %function
        .weak   wref, wdef
        .globl  wref, wdef
        .type   .Ltyped, %function
        .hidden .Lhid
        .data
wdef:   .long   wref
