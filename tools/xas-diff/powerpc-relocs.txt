=== DS- and DQ-form displacements against a symbol in 32-bit code
# The DS relocations belong to PowerPC64; GNU as writes the plain halfword ones
# here, where llvm-mc writes numbers R_PPC_* does not define. The alignment is
# spelled out, since GNU as leaves `.text` at 1 and llvm-mc, which rsasm
# follows, gives it 4.
        .text
        .p2align 2
        lxsd    3, ext(4)
        lxv     35, ext@l(4)
        stxssp  3, ext@l(4)
        stxv    3, dat(4)
        .data
dat:    .long 0
=== .gnu_attribute records the floating-point ABI
        .gnu_attribute 4, 1
        .text
        .globl  f
f:      pnop
        nop
=== .gnu_attribute tags are written in order
        .gnu_attribute 12, 1
        .gnu_attribute 8, 2
        .gnu_attribute 4, 1
        .text
        .globl  f
f:      pnop
        nop
=== thread-local data words, which llvm-mc writes as plain addresses
# GNU as writes R_PPC_DTPMOD32 and R_PPC_TPREL32 for these, and llvm-mc 22
# R_PPC_ADDR32, dropping the modifier; `@dtprel` the two agree on.
        .text
        .p2align 2
        nop
        .data
        .long   tv@dtpmod
        .long   tv@tprel
        .long   ext@dtpmod
        .long   ext+4@tprel
        .long   tv@dtprel
        .section .tbss,"awT",@nobits
tv:     .space  8
=== thread-local offsets in DS- and DQ-form fields in 32-bit code
# The plain halfword relocations, as for an address. llvm-mc writes the
# numbers of R_PPC_TLSGD and R_PPC_TLSLD for `@tprel` and `@tprel@l`.
        .text
        .p2align 2
        lxv     35, tv@tprel(3)
        lxv     35, tv@tprel@l(3)
        lxsd    2, tv@dtprel(3)
        stxsd   2, tv@dtprel@l(3)
        .section .tbss,"awT",@nobits
tv:     .space  8
=== the markers leave the symbol they name as it was
# GNU as leaves a symbol that only `@tls` or a call's `(sym@tlsgd)` names
# untyped; llvm-mc makes it thread-local. A local label outside a
# thread-local section is named as itself, where llvm-mc names its section.
        .text
        .p2align 2
        add     9, 9, ext@tls
        bl      __tls_get_addr(tv@tlsgd)
        bl      __tls_get_addr(ext2@tlsld)
        bl      __tls_get_addr(ext3+4@tlsgd)
lab:    add     9, 9, lab@tls
        .section .tbss,"awT",@nobits
tv:     .space  8
=== refused: a thread-local model on a variable defined outside a thread-local section
        .text
        .p2align 2
y:      addi    3, 3, y@tprel
=== refused: a thread-local GOT entry for a symbol plus an offset, in 32-bit code
        .text
        .p2align 2
        addi    3, 31, tv+4@got@tprel
        .section .tbss,"awT",@nobits
tv:     .space  8
=== refused: an offset after a call's thread-local argument
        .text

        bl      __tls_get_addr(tv@tlsgd+4)
        .section .tbss,"awT",@nobits
tv:     .space  8
