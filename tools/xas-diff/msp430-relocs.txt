# Whole objects, compared by tools/mc-diff/canon.sh: each reference form
# through each kind of symbol. GNU as relocates every reference from code on
# this target, even to the label before it, and names a code label itself
# rather than its section.
#
# Three things GNU as does that rsasm does not copy are kept out, none of
# which changes what a linker makes of the object:
#
# - the addend of the last R_MSP430_SYM_DIFF in a section, which GNU as
#   writes as the subtrahend's negated value where it writes zero for every
#   other, so a difference is never the last relocation in a section here;
# - where a `.section .data` directive, which refers to `__crt0_movedata` as
#   it is read, falls among the undefined symbols, since rsasm orders those
#   by where they are relocated;
# - a number `.set` after the code that uses it, which GNU as relocates
#   against the symbol and rsasm writes into the field.
=== an undefined symbol
        .text
        mov     X, r5
        mov     &X, r5
        mov     #X, r5
        mov     X(r4), r5
        mov     r5, X
        mov     r5, &X
        mov     r5, X(r6)
        mov.b   X, r5
        mov.b   &X, r5
        mov     &X, X
        mov     @r4, X
        mov     X, &X
        add     #X, X(r7)
        mov     #lo(X), r5
        mov     #hi(X), r5
        jmp     X
        jz      X
        call    #X
        call    X
        call    &X
        br      X
        br      #X
        br      &X
        br      X(r5)
        push    #X
