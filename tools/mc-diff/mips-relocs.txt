=== references to symbols in their own section, by binding
        .set    noreorder
        .text
        .globl  glob, hid, prot, intl, galias, func
        .hidden hid
        .protected prot
        .internal intl
        .weak   weak1, walias, whid
        .hidden whid
        .type   func, %function
        .hidden lhid
start:
        bal loc
        bal glob
        bal weak1
        bal hid
        bal prot
        bal intl
        bal whid
        bal lhid
        bal func
        bal late
        bal lweak
        bal lalias
        bal lwalias
        bal galias
        bal walias
        b loc
        b glob
        b weak1
        b lalias
        b walias
        beq $4, $5, loc
        beq $4, $5, glob
        beq $4, $5, weak1
        beq $4, $5, lalias
        beq $4, $5, walias
        nop
loc:    nop
glob:   nop
weak1:  nop
hid:    nop
prot:   nop
intl:   nop
whid:   nop
lhid:   nop
func:   nop
late:   nop
lweak:  nop
        .globl  late
        .weak   lweak
        .set    lalias, glob
        .set    lwalias, weak1
        .set    galias, loc
        .set    walias, loc

=== references back to symbols in their own section
        .set    noreorder
        .text
        .globl  glob
        .weak   weak1
loc:    nop
glob:   nop
weak1:  nop
        .set    lalias, glob
        bal loc
        bal glob
        bal weak1
        bal lalias
        b loc
        b glob
        b weak1
        beq $4, $5, loc
        beq $4, $5, glob
        beq $4, $5, weak1

=== references out of their section, by binding
        .set    noreorder
        .text
        .globl  oglob
        .weak   oweak, uweak
        .set    oalias, oloc
        bal oloc
        bal oglob
        bal oweak
        bal oalias
        bal ext
        bal uweak
        nop
        .section .text.other,"ax"
oloc:   nop
oglob:  nop
oweak:  nop

# `jalx` gets the same R_MIPS_26 as `jal`: what the linker has to do differs
# only in the opcode, which already says the call changes ISA mode. Both
# targets here are undefined, because rsasm assembles neither MIPS16 nor
# microMIPS and GNU ld refuses a JALX that resolves to plain MIPS code
# ("unsupported JALX to the same ISA mode").
=== a call that changes ISA mode
        .set    noreorder
        .text
        .weak   uweak
        jalx    ext
        jalx    uweak
        nop

=== data referring to symbols, by binding
        .set    noreorder
        .data
        .globl  glob, galias, hid
        .weak   weak1, walias
        .weak   uweak
        .hidden hid
        .set    lalias, glob
        .set    lwalias, weak1
        .set    galias, loc
        .set    walias, loc
base:   .long   loc, glob, weak1, hid, lalias, lwalias, galias, walias, ext
        .long   loc - ., glob - ., weak1 - ., lalias - ., walias - .
        .long   glob - base, weak1 - base, base - weak1, lalias - base, walias - base
        .long   ext - ., uweak - .
loc:    .long   0
glob:   .long   0
weak1:  .long   0
hid:    .long   0

=== a double-precision operand names a register pair
        .set    noreorder
        .globl  f
f:      add.d   $f4, $f6, $f8
        div.d   $f10, $f12, $f14
        mov.d   $f16, $f18
        neg.d   $f20, $f22
        sqrt.d  $f24, $f26
        c.eq.d  $f28, $f30
        c.ult.d $f0, $f2
        ldc1    $f4, 8($sp)
        sdc1    $f6, 16($a0)

=== single precision and the integer formats name one register each
        .set    noreorder
        .globl  f
f:      add.s   $f4, $f6, $f8
        mov.s   $f10, $f11
        abs.s   $f12, $f13
        c.eq.s  $f14, $f15
        lwc1    $f16, 8($sp)
        swc1    $f17, 16($a0)
        mtc1    $4, $f18
        mfc1    $5, $f19
        round.w.s $f20, $f21

=== a conditional move on a flag pairs what its format says
        .set    noreorder
        .globl  f
f:      movt.d  $f4, $f8, $fcc1
        movf.d  $f12, $f16, $fcc7
        movt.s  $f20, $f21, $fcc0
        movf.s  $f22, $f23, $fcc5

=== a mixed-format conversion pairs only the operands that hold a double
        .set    noreorder
        .globl  f
f:      cvt.d.s $f4, $f6
        cvt.s.d $f8, $f10
        cvt.d.w $f12, $f14
        cvt.w.d $f16, $f18
        trunc.w.d $f20, $f22
        floor.w.d $f24, $f26
        ceil.w.d $f28, $f30

=== .module fp=64 puts a double in one register
        .module fp=64
        .set    noreorder
        .globl  f
f:      add.d   $f4, $f5, $f6
        mov.d   $f7, $f8
        c.eq.d  $f9, $f10
        ldc1    $f11, 8($sp)
        sdc1    $f12, 16($a0)
        cvt.d.s $f13, $f14
        trunc.w.d $f15, $f16

# rsasm follows GNU as for common blocks, and so parts from llvm-mc on
# what these leave out: a `.comm` that names no alignment, which GNU as
# aligns by its size and llvm-mc to 1; what `.lcomm` reserves, which
# llvm-mc packs; and a symbol only `.size`, `.local` or `.weak` names (see
# the x86 corpora in tools/gas-diff, and tests/output.rs). A `.L` name
# only a directive names is left out too, as GNU as leaves it out; llvm-mc
# writes it for o32, where its own local labels start with `$` instead.
=== a .local .comm with its alignment, reserved in .bss
        .local  b8, b4
        .comm   b8, 8, 8
        .comm   b4, 3, 4
        .data
        .word   b8, b4

        .type   tobj, @object
        .hidden hid
        .protected prot
        .internal intl
        .globl  gonly
