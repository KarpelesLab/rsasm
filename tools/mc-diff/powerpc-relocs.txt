=== references to symbols in their own section, by binding
        .text
        .globl  glob, hid, prot, intl, galias, func
        .hidden hid
        .protected prot
        .internal intl
        .weak   weak1, walias, whid
        .hidden whid
        .type   func, %function
        .hidden lhid
start:
        bl loc
        bl glob
        bl weak1
        bl hid
        bl prot
        bl intl
        bl whid
        bl lhid
        bl func
        bl late
        bl lweak
        bl lalias
        bl lwalias
        bl galias
        bl walias
        b loc
        b glob
        b weak1
        b lalias
        b walias
        beq loc
        beq glob
        beq weak1
        beq lalias
        beq walias
        nop
loc:    nop
glob:   nop
weak1:  nop
hid:    nop
prot:   nop
intl:   nop
whid:   nop
lhid:   nop
func:   nop
late:   nop
lweak:  nop
        .globl  late
        .weak   lweak
        .set    lalias, glob
        .set    lwalias, weak1
        .set    galias, loc
        .set    walias, loc

=== references back to symbols in their own section
        .text
        .globl  glob
        .weak   weak1
loc:    nop
glob:   nop
weak1:  nop
        .set    lalias, glob
        bl loc
        bl glob
        bl weak1
        bl lalias
        b loc
        b glob
        b weak1
        beq loc
        beq glob
        beq weak1

        .set    oalias, oloc

        bl oweak
        bl oalias
        bl ext

        nop
        .section .text.other,"ax"
oloc:   nop
oglob:  nop
oweak:  nop

=== calls through and around the PLT, and GOT entries
        .text
        .globl  glob
        .weak   weak1
start:
        bl      ext@plt
        b       ext@plt
        bl      glob@plt
        bl      weak1@plt
        bl      loc@plt
        bl      ext@local
        b       glob@local
        bl      loc@local
        lis     3, ext@got@ha

        addi    4, 4, ext@got
        addi    5, 5, glob@got@h

glob:   nop
weak1:  nop

# rsasm follows GNU as for common blocks, and so parts from llvm-mc on
# what these leave out: a `.comm` that names no alignment, which GNU as
# aligns by its size and llvm-mc to 1; what `.lcomm` reserves, which
# llvm-mc packs; and a symbol only `.size`, `.local` or `.weak` names (see
# the x86 corpora in tools/gas-diff, and tests/output.rs).
=== a .local .comm with its alignment, reserved in .bss
        .local  b8, b4
        .comm   b8, 8, 8
        .comm   b4, 3, 4
        .data
        .long   b8, b4
