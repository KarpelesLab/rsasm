=== backward branch to a named label
start:
        mov     r0, 0
        add     r0, r0, 1
        b       start
=== forward branch
        b       done
        mov     r0, 1
        mov     r1, 2
done:
        nop
=== conditional branches around a loop
loop:
        subs    r0, r0, 1
        bne     loop
        bgt     out
        cmp     r0, 0
        beq     loop
out:
        nop
=== numeric local labels
1:      add     r0, r0, 1
        cmp     r0, 16
        blt     1b
        b       2f
        nop
2:      nop
=== branch to an absolute expression
        b       . + 8
        nop
        nop
        b       . - 4
=== switching to thumb and back
        add     r0, r1, r2
        .thumb
        adds    r0, r1, r2
        movs    r3, 7
        .arm
        add     r0, r1, r2
=== mixing code and data
        mov     r0, 1
        .long   0x12345678
        .byte   1, 2, 3, 4
        mov     r1, 2
=== code 16 and code 32 select the instruction set
        add     r0, r1, r2
        .code   16
        adds    r0, r1, r2
        movs    r3, 7
        .code   32
        add     r0, r1, r2
=== alignment padding in arm code
        mov     r0, r0
        .balign 16
        mov     r0, r0
=== a literal pool entry holding a difference of labels
        @ GNU as refuses this one: `parse_big_immediate` takes a constant, a
        @ bignum or a symbol plus an addend, and a difference of two labels
        @ that is not already a number is none of the three, so `ldr r0,
        @ =l1-l0` is a syntax error there. llvm-mc writes the entry, and so
        @ does rsasm; see the ARM backend's module documentation.
l0:     nop
        ldr     r0, =l1-l0
        bx      lr
        .space  1000
l1:     nop
        .ltorg
=== a load that names its label
back:   .word   0x12345678
        ldr     r0, back
        ldrb    r1, back
        ldrh    r2, back
        ldrsb   r3, back
        ldrsh   r4, back
        ldrd    r6, r7, back
        str     r0, back
        strh    r2, back
        ldr     r0, fwd
        pld     back
fwd:    bx      lr
=== a coprocessor transfer that names its label
        @ The offset counts words in eight bits, so it reaches a quarter of
        @ what `ldr` does. llvm-mc writes the same bytes as GNU as for a
        @ label in the section.
back:   .word   0x12345678
        vldr    s0, back
        vldr    d1, back
        vstr    s2, back
        ldc     p1, c2, back
        stc     p1, c3, back
        ldc2    p1, c4, back
        vldr    s5, fwd
fwd:    bx      lr
