=== references to symbols in their own section, by binding
        .text
        .globl  glob, hid, prot, intl, galias, func
        .hidden hid
        .protected prot
        .internal intl
        .weak   weak1, walias, whid
        .hidden whid
        .type   func, %function
        .hidden lhid
start:
        bl loc
        bl glob
        bl weak1
        bl hid
        bl prot
        bl intl
        bl whid
        bl lhid
        bl func
        bl late
        bl lweak
        bl lalias
        bl lwalias
        bl galias
        bl walias
        b loc
        b glob
        b weak1
        b lalias
        b walias
        b.eq loc
        b.eq glob
        b.eq weak1
        b.eq lalias
        b.eq walias
        adr x0, loc
        adr x0, glob
        adr x0, weak1
        adr x0, lalias
        adr x0, galias
        nop
loc:    nop
glob:   nop
weak1:  nop
hid:    nop
prot:   nop
intl:   nop
whid:   nop
lhid:   nop
func:   nop
late:   nop
lweak:  nop
        .globl  late
        .weak   lweak
        .set    lalias, glob
        .set    lwalias, weak1
        .set    galias, loc
        .set    walias, loc

=== references back to symbols in their own section
        .text
        .globl  glob
        .weak   weak1
loc:    nop
glob:   nop
weak1:  nop
        .set    lalias, glob
        bl loc
        bl glob
        bl weak1
        bl lalias
        b loc
        b glob
        b weak1
        b.eq loc
        b.eq glob
        b.eq weak1

=== references out of their section, by binding
        .text
        .globl  oglob
        .weak   oweak, uweak
        .set    oalias, oloc
        bl oloc
        bl oglob
        bl oweak
        bl oalias
        bl ext
        bl uweak
        nop
        .section .text.other,"ax"
oloc:   nop
oglob:  nop
oweak:  nop

=== data referring to symbols, by binding
        .data
        .globl  glob, galias, hid
        .weak   weak1, walias
        .weak   uweak
        .hidden hid
        .set    lalias, glob
        .set    lwalias, weak1
        .set    galias, loc
        .set    walias, loc
base:   .long   loc, glob, weak1, hid, lalias, lwalias, galias, walias, ext
        .long   loc - ., glob - ., weak1 - ., lalias - ., walias - .
        .long   glob - base, weak1 - base, base - weak1, lalias - base, walias - base
        .long   ext - ., uweak - .
loc:    .long   0
glob:   .long   0
weak1:  .long   0
hid:    .long   0

=== move-wide groups of a symbol's address
# An address is built sixteen bits at a time, and the operator picks which
# sixteen. A label reaches the linker whichever section it is in, even for
# `:prel_gN:`, whose PC-relative arithmetic the relocation is what asks for;
# only a constant is taken here.
#
# `movk` with a signed group -- `:abs_gN_s:` and every `:prel_gN:` but the
# `_nc` ones -- is not here: llvm-mc takes it, GNU as refuses it because the
# linker may have to write such a value as its inverse, which only `movz` and
# `movn` can carry, and rsasm follows GNU as.
        .text
        .globl  glob
        .weak   weak1
        .set    kabs, 0x123456789abcdef0
        .set    ksmall, -0x1234
f:
        movz    x0, :abs_g0_nc:glob
        movk    x0, :abs_g1_nc:glob
        movk    x0, :abs_g2_nc:glob
        movk    x0, :abs_g3:glob
        movz    x1, :abs_g0:ext
        movz    x2, :abs_g1:ext + 16
        movz    x3, :abs_g2:weak1
        movn    x4, :abs_g0_s:ext
        movz    x5, :abs_g1_s:ext - 8
        movn    x6, :abs_g2_s:ext
        movz    x7, :prel_g0:ext
        movk    x7, :prel_g0_nc:ext
        movz    x8, :prel_g1:loc
        movk    x8, :prel_g1_nc:loc
        movz    x9, :prel_g2:dat
        movk    x9, :prel_g2_nc:dat
        movz    x10, :prel_g3:dat
        movz    w11, :abs_g0_nc:glob
        movk    w11, :abs_g1_nc:glob
        movz    w12, :prel_g0:ext - 8
        movz    x13, :abs_g0_nc:kabs
        movk    x13, :abs_g3:kabs
        movz    x14, :abs_g0_s:ksmall
loc:    nop
glob:   nop
weak1:  nop
        .data
dat:    .quad   0

=== thread-local access models
# Only the linker computes any of these, and it may rewrite a whole sequence
# into another model, so each is relocated whatever it refers to: by name,
# even for a variable in this file, and an undefined target becomes
# `STT_TLS`.
#
# Left out, and in tools/xas-diff's corpus against GNU as, whom rsasm
# follows: `:tlsgd:`, `:tlsldm:`, their `_lo12` and move-wide forms and
# `:tlsdesc_off_g1:`, which llvm-mc 22 does not know; `movz` with a signed
# group, which llvm-mc makes `movn` and GNU as leaves alone (so the signed
# groups are written with `movn` here); `:dtprel_hi12:`, whose `add` llvm-mc
# shifts by twelve and GNU as only where `lsl #12` follows; an `add` operator
# followed by `lsl #0` or `lsl #12`, which llvm-mc refuses; `movk` with
# `:tprel_g1:` or `:tprel_g0:`, which GNU as refuses; `:tlsdesc:` on `adr` or
# a literal load, which llvm-mc refuses or relocates as a plain literal;
# `:gottprel_lo12:` and `:tlsdesc_lo12:` on a load narrower than 64 bits,
# which llvm-mc refuses; and a 128-bit access under `:tprel_lo12:` or
# `:dtprel_lo12:`, which llvm-mc relocates and GNU as refuses.
        .section .tbss,"awT",@nobits
        .p2align 3
tb:     .space  8
        .globl  tg
tg:     .space  8
        .text
        .globl  f
f:
        movn    x0, :tprel_g2:v
        movk    x0, :tprel_g1_nc:v
        movk    x0, :tprel_g0_nc:v
        movn    x1, :tprel_g1:v
        movn    w2, :tprel_g0:v
        movz    x3, :tprel_g1_nc:tb
        mrs     x4, tpidr_el0
        add     x4, x4, :tprel_hi12:v
        add     x4, x4, :tprel_lo12_nc:v
        add     x5, x5, :tprel_lo12:tg+4
        ldrb    w6, [x4, :tprel_lo12:v]
        ldrh    w6, [x4, :tprel_lo12_nc:v]
        ldr     w6, [x4, :tprel_lo12:v]
        ldr     x6, [x4, :tprel_lo12_nc:v]
        str     d6, [x4, :tprel_lo12:v]
        movn    x7, :dtprel_g2:v
        movk    x7, :dtprel_g1:v
        movk    x7, :dtprel_g0_nc:v
        add     x8, x8, :dtprel_lo12_nc:v
        ldr     x8, [x8, :dtprel_lo12:v]
        adrp    x9, :gottprel:v
        ldr     x9, [x9, :gottprel_lo12:v]
        ldr     x10, :gottprel:tg
        movn    x11, :gottprel_g1:v
        movk    x11, :gottprel_g0_nc:v
        adrp    x0, :tlsdesc:v
        ldr     x1, [x0, :tlsdesc_lo12:v]
        add     x0, x0, :tlsdesc_lo12:v
        .tlsdesccall v
        blr     x1
