# llvm-mc packs; and a symbol only `.size`, `.local` or `.weak` names (see

        .globl  f

        addi    a0, a0, %tprel_lo(tv)
        lui     a1, %tprel_hi(tg+4)

        lw      a2, %tprel_lo(tg+4)(a1)
        sw      a2, %tprel_lo(tg+4)(a1)

        addi    a3, a3, %tprel_lo(ext)
