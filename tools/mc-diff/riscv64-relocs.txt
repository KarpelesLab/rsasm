=== la and lla of an external symbol relocate both halves
        la      a0, ext
        lla     a1, ext
        la      a2, ext+4
        lla     a3, ext-8

=== la of a label in another section
        la      a0, value
        lla     a1, value+2
        .data
        .word   0
value:  .word   1

=== la of a label in the same section resolves
back:   nop
        la      a0, back
        lla     a1, ahead
        la      a2, 1f
1:
ahead:  ret

=== la of a plain number is li
        .equ    K, 0x12345
        la      a0, 0x1000
        lla     a1, K
        la      a2, -1

=== la of a symbol set to a label
        .set    alias, value+4
        la      a0, alias
        .data
value:  .dword  0

=== pairs in a data section
        .data
        .word   1
        lla     a0, ext
        la      a1, 2f
2:      .word   2

=== the pair label follows a branch that grew
        beqz    a0, far
        .space  300
far:    la      a0, ext
        lw      a1, ext

=== pairs with compression off
        .option norvc
        la      a0, ext
        lw      a1, ext
        .option rvc
        la      a2, ext

=== integer loads from a symbol
        lb      a0, ext
        lbu     a1, ext
        lh      a2, ext
        lhu     a3, ext
        lw      a4, ext
        lwu     a5, ext
        ld      s1, ext+16

=== integer stores to a symbol
        sb      a0, ext, t0
        sh      a1, ext, t1
        sw      a2, ext, t2
        sd      a3, ext+8, s1

=== floating-point loads and stores through a symbol
        flw     fa0, ext, t0
        fld     fa1, ext, t1
        fsw     fa2, ext, t2
        fsd     fa3, ext, a0

=== loads and stores of a local label
        lw      a0, near
        sd      a0, near, t0
        ld      a1, far
        fld     fa0, far, t2
near:   .dword  0
        .data
far:    .dword  0

=== lga goes through the GOT
        lga     a0, ext
        lga     a1, ext+8
        lga     a2, here
        lga     a3, 1f
here:   nop
1:      nop

=== option pic sends la through the GOT
        la      a0, ext
        .option pic
        la      a1, ext
        la      a2, local
        lla     a3, ext
        la      a4, 0x800
        .option nopic
        la      a5, ext
local:  ret

=== option push and pop keep pic
        .option push
        .option pic
        .option push
        .option norvc
        la      a0, ext
        .option pop
        la      a1, ext
        .option pop
        la      a2, ext
        mv      a3, a4

=== call, tail and jump
        call    ext
        call    ext@plt
        tail    ext
        call    t0, ext
        jump    ext, t1

=== local call, tail and jump resolve
        call    target
        tail    target
        jump    target, t0
target: ret
        call    target
        jump    target, a5

=== call of a label in another section
        call    far
        tail    far
        .section .text.far,"ax",@progbits
far:    ret

=== pcrel_hi and pcrel_lo name the auipc label
foo:    auipc   a0, %pcrel_hi(ext)
        addi    a0, a0, %pcrel_lo(foo)
        ld      a1, %pcrel_lo(foo)(a0)
        sd      a1, %pcrel_lo(foo)(a0)

=== pcrel_lo of numeric labels
1:      auipc   a0, %pcrel_hi(ext)
        addi    a0, a0, %pcrel_lo(1b)
1:      auipc   a1, %pcrel_hi(value)
        lw      a1, %pcrel_lo(1b)(a1)
        sw      a1, %pcrel_lo(1b)(a1)
        .data
value:  .word   0

        .data
value:  .dword  0
