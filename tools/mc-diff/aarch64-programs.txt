=== switch table
dispatch:
        adr     x1, table
        ldrsw   x2, [x1, x0, lsl 2]
        add     x1, x1, x2
        br      x1
table:
        # `.4byte`, not `.word`: GNU as makes `.word` four bytes on AArch64 and
        # two on x86, and rsasm's core currently uses the x86 size everywhere.
        .4byte  case0 - table
        .4byte  case1 - table
case0:
        mov     w0, 0
        ret
case1:
        mov     w0, 1
        ret

=== adrp with its :lo12: and :got: companions
# Every field here is filled by the linker, so the bytes are the bare opcodes;
# the relocation types are checked in tests/aarch64.rs.
        .text
f:
        adrp    x0, var
        add     x0, x0, :lo12:var
        ldr     x2, [x0, :lo12:var]
        ldr     w3, [x0, :lo12:var]
        ldrb    w4, [x0, :lo12:var]
        ldrh    w5, [x0, :lo12:var]
        ldr     q6, [x0, :lo12:var]
        str     x7, [x0, :lo12:var]
        adrp    x8, :got:var
        ldr     x9, [x8, :got_lo12:var]
        add     x10, x10, :lo12:local
local:
        nop

=== compare and branch against a number (FEAT_CMPBR)
# The inclusive comparisons move the number rather than the condition,
# so `cbge x0, #1` is `cbgt x0, #0` and its range runs to 64.
head:
        cbgt    x0, #0, tail
        cbgt    w5, #63, head
        cbgt    xzr, #31, tail
        cbge    x0, #1, tail
        cbge    w5, #64, head
        cbge    xzr, #32, tail
        cblt    x0, #0, tail
        cblt    w5, #63, head
        cblt    xzr, #31, tail
        cble    x0, #-1, tail
        cble    w5, #62, head
        cble    xzr, #30, tail
        cbhi    x0, #0, tail
        cbhi    w5, #63, head
        cbhi    xzr, #31, tail
        cbhs    x0, #1, tail
        cbhs    w5, #64, head
        cbhs    xzr, #32, tail
        cblo    x0, #0, tail
        cblo    w5, #63, head
        cblo    xzr, #31, tail
        cbls    x0, #-1, tail
        cbls    w5, #62, head
        cbls    xzr, #30, tail
        cbeq    x0, #0, tail
        cbeq    w5, #63, head
        cbeq    xzr, #31, tail
        cbne    x0, #0, tail
        cbne    w5, #63, head
        cbne    xzr, #31, tail
tail:
        ret
