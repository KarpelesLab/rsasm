=== the relocations the V9 branches need
# `BPcc`, `FBPfcc` and `BPr` narrow the displacement to make room for the
# bank and the prediction hint, so each has a relocation of its own that
# `Bicc` and `FBfcc` do not use.
        .text
        .globl  glob
        .weak   weak1
start:
        be      glob
        be      %icc, glob
        be      %xcc, weak1
        brz     %g1, glob
        brnz    %g1, ext
        fbe     glob
        fbe     %fcc0, glob
        fbo     %fcc3, weak1
        fbne    ext
        fbne    %fcc2, ext
        nop
glob:   nop
weak1:  nop

=== backwards to a label in the same section
        .text
back:
        nop
        be      %icc, back
        brz     %g1, back
        fbe     back
        fbe     %fcc1, back

# rsasm follows GNU as for common blocks, and so parts from llvm-mc on
# what these leave out: a `.comm` that names no alignment, which GNU as
# aligns by its size and llvm-mc to 1; what `.lcomm` reserves, which
# llvm-mc packs; and a symbol only `.size`, `.local` or `.weak` names (see
# the x86 corpora in tools/gas-diff, and tests/output.rs).
=== a .local .comm with its alignment, reserved in .bss
        .local  b8, b4
        .comm   b8, 8, 8
        .comm   b4, 3, 4
        .data
        .xword  b8, b4
