=== short backward branch
start:
        movs    r0, 0
        adds    r0, 1
        b       start
=== short forward branch
        b       done
        movs    r0, 1
done:
        nop
=== conditional branches
loop:
        subs    r0, 1
        bne     loop
        beq     out
        cmp     r0, 0
        bgt     loop
out:
        nop
=== bl to a local label
target:
        nop
        bl      target
=== explicitly wide branches
here:
        b.w     here
        bne.w   here
        nop
=== branches that do fit the 16-bit form
near:
        nop
        b       near
        bne     near
=== a branch too far for the 16-bit form
far:
        nop
        .space  4096
        b       far
        bne     far
=== numeric local labels
1:      adds    r0, 1
        cmp     r0, 16
        blt     1b
        b       2f
        nop
2:      nop
=== starting in thumb and switching to arm on a word boundary
        adds    r0, r1, r2
        movs    r3, 7
        .arm
        add     r0, r1, r2
=== alignment padding in thumb code
        movs    r0, r0
        @ A two-byte gap is a 16-bit `nop` in both. A longer one is not: GNU
        @ as, which rsasm follows for ARM, writes 32-bit `nop.w`s and only as
        @ many 16-bit ones as the gap's low bits need (its `arm_handle_align`),
        @ where llvm-mc writes 16-bit no-ops throughout. See
        @ tools/xas-diff/thumb.txt.
        .balign 4
        movs    r0, r0
        @ GNU as, and rsasm, pad the end of a code section to its alignment
        @ up to a word; llvm-mc does not. Ending on a word keeps that out of
        @ this case.
        movs    r0, r0
=== a load that names its label
        @ The 16-bit form reaches a word-aligned label up to 1020 bytes
        @ ahead; everything else is 32 bits wide. llvm-mc sizes these the
        @ way GNU as does, which is what rsasm follows.
back:   .word   0x12345678
        ldr     r0, back
        ldrb    r1, back
        ldrh    r2, back
        ldrsh   r3, back
        ldrd    r6, r7, back
        ldr     r8, back
        pld     back
        .p2align 2, 0
        ldr     r0, fwd
        ldrh    r1, fwd
        nop
        .p2align 2, 0
fwd:    bx      lr
        nop
=== a coprocessor transfer that names its label
        @ The PC is rounded down to a word, as it is for the loads above.
back:   .word   0x12345678
        vldr    s0, back
        vldr    d1, back
        ldc     p1, c2, back
        stc     p1, c3, back
        ldc2    p1, c4, back
        nop
        .p2align 2, 0
        vldr    s5, fwd
        nop
        .p2align 2, 0
fwd:    bx      lr
        nop
=== a coprocessor transfer of coprocessor 9
        @ GNU as reads any pre-indexed T32 transfer of coprocessor 9 as the
        @ half-precision `vldr`, whose offset counts halfwords, and writes
        @ half the field the architecture asks for; llvm-mc counts words,
        @ and so does rsasm. See the ARM backend's module documentation.
back:   .word   0
        ldc     p9, c1, back
        ldc     p9, c1, [r0, #8]
        stc2l   p9, c1, back
