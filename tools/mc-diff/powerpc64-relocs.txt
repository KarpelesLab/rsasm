=== references to symbols in their own section, by binding
        .text
        .globl  glob, hid, prot, intl, galias, func
        .hidden hid
        .protected prot
        .internal intl
        .weak   weak1, walias, whid
        .hidden whid
        .type   func, %function
        .hidden lhid
start:
        bl loc
        bl glob
        bl weak1
        bl hid
        bl prot
        bl intl
        bl whid
        bl lhid
        bl func
        bl late
        bl lweak
        bl lalias
        bl lwalias
        bl galias
        bl walias
        b loc
        b glob
        b weak1
        b lalias
        b walias
        beq loc
        beq glob
        beq weak1
        beq lalias
        beq walias
        nop
loc:    nop
glob:   nop
weak1:  nop
hid:    nop
prot:   nop
intl:   nop
whid:   nop
lhid:   nop
func:   nop
late:   nop
lweak:  nop
        .globl  late
        .weak   lweak
        .set    lalias, glob
        .set    lwalias, weak1
        .set    galias, loc
        .set    walias, loc

=== references back to symbols in their own section
        .text
        .globl  glob
        .weak   weak1
loc:    nop
glob:   nop
weak1:  nop
        .set    lalias, glob
        bl loc
        bl glob
        bl weak1
        bl lalias
        b loc
        b glob
        b weak1
        beq loc
        beq glob
        beq weak1

=== references out of their section, by binding
        .text
        .globl  oglob
        .weak   oweak, uweak
        .set    oalias, oloc
        bl oloc
        bl oglob
        bl oweak
        bl oalias
        bl ext
        bl uweak
        nop
        .section .text.other,"ax"
oloc:   nop
oglob:  nop
oweak:  nop

=== data referring to symbols, by binding
        .data

        .weak   weak1, walias
        .weak   uweak
        .hidden hid
        .set    lalias, glob
        .set    lwalias, weak1

        .set    walias, loc
base:   .long   loc, glob, weak1, hid, lalias, lwalias, galias, walias, ext

        .long   glob - base, weak1 - base, base - weak1, lalias - base, walias - base
        .long   ext - ., uweak - .

glob:   .long   0
weak1:  .long   0
hid:    .long   0

        sldi    4, 4, 32
