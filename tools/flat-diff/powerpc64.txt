Lines before the first snippet are not assembled.

Sections start with `.p2align 4`, which is llvm-mc's default alignment for
PowerPC code and data and not rsasm's; see mips.txt.

=== ha and l of a label in another section
	.text
	.p2align 4
	lis 3, msg@ha
	addi 3, 3, msg@l
	lwz 4, msg@l(3)
	ld 5, msg+8@l(3)
	std 5, msg+8@l(3)
	blr
	.data
	.p2align 4
	.space 0x8010
msg:	.quad 1, 2

=== h and ha differ where bit 15 is set @ 0x10007ff0
	.text
	.p2align 4
here:
	lis 3, there@h
	ori 3, 3, there@l
	lis 4, there@ha
	addi 4, 4, there@l
	blr
there:
	blr

=== ha and l with an addend
	.text
	.p2align 4
	lis 3, table+0x10000@ha
	addi 3, 3, table+0x10000@l
	lis 4, table-8@ha
	addi 4, 4, table-8@l
	blr
	.data
	.p2align 4
table:	.quad 0

=== branches into another section
	.text
	.p2align 4
entry:
	bl helper
	b helper
	beq helper
	.section .text.helper, "ax"
	.p2align 4
helper:
	b entry
	bdnz entry

=== data across sections
	.text
	.p2align 4
entry:
	blr
	.data
	.p2align 4
	.quad entry
	.long entry - .
	.quad entry - .
