=== calls and branches between objects
--- a.s
        .text
        .globl  _start
        .type   _start, @function
_start:
        call    helper
        call    weakfn
        call    absent
        call    helper@PLT
        jmp     helper
        je      helper
        jne     weakfn
        ret
        .weak   absent
--- b.s
        .text
        .globl  helper
        .weak   weakfn
        .type   helper, @function
helper:
        call    _start
        jmp     _start
        loop    helper
        ret
weakfn:
        ret

=== data references between objects
--- a.s
        .data
        .globl  dptr
        .type   dptr, @object
dptr:   .long   target
        .long   target + 16
        .long   target - 8
        .long   target - .
        .long   wsym
        .long   com1
        .long   bssvar
        .long   after - dptr
        .size   dptr, . - dptr
after:
        .text
        .globl  _start
_start: movl    $dptr, %eax
        movl    bssvar, %ecx
        movl    com1, %edx
        ret
--- b.s
        .text
        .globl  target
        .type   target, @function
target: nop
        .weak   wsym
wsym:   nop
        .comm   com1, 8, 8
        .bss
        .globl  bssvar
        .type   bssvar, @object
bssvar: .space  16
        .size   bssvar, 16

=== the GOT and the position-independent idiom
--- a.s
        .text
        .globl  _start
_start:
        call    __x86.get_pc_thunk.bx
        addl    $_GLOBAL_OFFSET_TABLE_, %ebx
        movl    shared@GOT(%ebx), %eax
        movl    shared@GOTOFF(%ebx), %ecx
        leal    shared@GOTOFF(%ebx), %edx
        call    helper@PLT
        ret
        .globl  __x86.get_pc_thunk.bx
__x86.get_pc_thunk.bx:
        movl    (%esp), %ebx
        ret
--- b.s
        .text
        .globl  helper
helper: ret
        .data
        .globl  shared
shared: .long   42

=== a weak definition overridden by a strong one in another object
--- a.s
        .text
        .globl  _start
_start:
        call    both
        call    weakonly
        movl    $both, %eax
        ret
        .weak   both
both:   nop
        nop
--- b.s
        .text
        .globl  both
both:   int3
        int3
        .weak   weakonly
weakonly:
        ret

=== common symbols merged between objects
--- a.s
        .comm   shared, 4, 4
        .comm   bigger, 8, 8
        .text
        .globl  _start
_start:
        movl    shared, %eax
        movl    bigger, %ecx
        movl    $shared, %edx
        ret
--- b.s
        .comm   shared, 4, 4
        .comm   bigger, 64, 32
        .text
        .globl  reader
reader:
        movl    $bigger, %eax
        movl    shared, %ecx
        ret

=== references into another object's sections
--- a.s
        .section .rodata
        .globl  table
table:  .long   entry0
        .long   entry1
        .long   entry0 - table
        .long   0
        .section .text.hot,"ax",@progbits
        .globl  _start
_start:
        call    entry0
        call    entry1
        movl    $table, %eax
        ret
--- b.s
        .section .text.hot,"ax",@progbits
        .globl  entry0, entry1
entry0: nop
        nop
entry1: ret
        .section .rodata
        .globl  msg
msg:    .asciz  "linked"

=== thread-local variables across objects
--- a.s
        .text
        .globl  _start
_start:
        # General dynamic, then local dynamic: the linker matches on the
        # instructions as well as the relocations when it turns these into
        # local-exec, so only these forms can be written.
        leal    tv@TLSGD(,%ebx,1), %eax
        call    ___tls_get_addr@PLT
        leal    tloc@TLSLDM(%ebx), %eax
        call    ___tls_get_addr@PLT
        leal    tloc@DTPOFF(%eax), %edx
        leal    tv2@DTPOFF(%eax), %ecx
        # Initial exec, then local exec in both its spellings.
        movl    %gs:0, %eax
        subl    tv2@GOTTPOFF(%ebx), %eax
        movl    %gs:0, %edx
        leal    tv3@TPOFF(%edx), %ecx
        leal    tloc@NTPOFF(%edx), %esi
        movl    $tb@NTPOFF, %edi
        # A TLS descriptor, whose `@TLSCALL` relocation covers no bytes.
        leal    tv3@TLSDESC(%ebx), %eax
        call    *tv3@TLSCALL(%eax)
        ret
        .section .tdata,"awT",%progbits
        .globl  tv
tv:     .long   0x11223344
tloc:   .long   99
--- b.s
        .section .tdata,"awT",%progbits
        .globl  tv2, tv3
tv2:    .long   2
tv3:    .long   3
        .section .tbss,"awT",%nobits
        .globl  tb
tb:     .space  8
        .text
        .globl  helper
helper: movl    %gs:0, %eax
        leal    tb@NTPOFF(%eax), %edx
        leal    tv@TPOFF(%eax), %ecx
        ret
        .data
        .long   tv@DTPOFF
        .long   tv@NTPOFF
