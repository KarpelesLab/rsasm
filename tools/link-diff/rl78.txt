# RL78 programs for tools/link-diff.
#
# Each program is linked with and without `--relax`. As on RX, a difference
# of two symbols in different sections is not here: GNU as writes it as a
# stack of `R_RL78_SYM`, `R_RL78_OPsub` and a store, which one fixup cannot
# express, so rsasm refuses it -- `src/arch/rl78/reloc.rs` says so.
#
# The symbol types are written `%function` rather than `@function`: `@`
# separates statements in RL78 source, as it does in GNU as, and rsasm's
# lexer ends the statement there, where GNU as's `.type` reads the rest of
# the line raw and takes the `@` as the type's sigil.

=== calls and branches between objects
--- a.s
        .text
        .globl  _start
        .type   _start, %function
_start:
        call    $!helper
        br      $!helper
        call    $!weakfn
        movw    ax, #helper
        movw    hl, #weakfn
        ret
--- b.s
        .text
        .globl  helper
        .weak   weakfn
        .type   helper, %function
helper:
        call    $!_start
        br      $!_start
        ret
weakfn:
        ret

=== absolute references to another object
--- a.s
        .text
        .globl  _start
_start:
        movw    ax, #target
        movw    hl, #bssvar
        movw    de, #rodata
        movw    bc, #target + 8
        mov     a, !target
        ret
--- b.s
        .text
        .globl  target
target: nop
        nop
        .section .rodata
        .globl  rodata
rodata: .long   0x11223344
        .bss
        .globl  bssvar
bssvar: .space  16

=== data references and differences between objects
--- a.s
        .data
        .globl  dptr
        .type   dptr, %object
dptr:   .long   target
        .long   target + 16
        .long   wsym
        .long   com1
        .long   bssvar
        .long   after - dptr
        .size   dptr, . - dptr
after:  .long   0
        .text
        .globl  _start
_start: ret
--- b.s
        .text
        .globl  target
        .type   target, %function
target: nop
        .weak   wsym
wsym:   nop
        .comm   com1, 8, 8
        .bss
        .globl  bssvar
        .type   bssvar, %object
bssvar: .space  16
        .size   bssvar, 16

=== a weak definition overridden by a strong one in another object
--- a.s
        .text
        .globl  _start
_start:
        call    $!both
        call    $!weakonly
        movw    ax, #both
        ret
        .weak   both
both:   nop
        nop
--- b.s
        .text
        .globl  both
both:   brk
        brk
        .weak   weakonly
weakonly:
        ret

=== common symbols merged between objects
--- a.s
        .comm   shared, 4, 4
        .comm   bigger, 8, 8
        .text
        .globl  _start
_start:
        movw    ax, #shared
        movw    hl, #bigger
        ret
--- b.s
        .comm   shared, 4, 4
        .comm   bigger, 64, 32
        .text
        .globl  reader
reader:
        movw    ax, #bigger
        ret
