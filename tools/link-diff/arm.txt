=== calls and branches between objects
--- a.s
        .text
        .globl  _start
        .type   _start, %function
_start:
        bl      helper
        b       helper
        bleq    helper
        bne     helper
        bl      weakfn
        bl      absent
        bx      lr
        .weak   absent
--- b.s
        .text
        .globl  helper
        .weak   weakfn
        .type   helper, %function
helper:
        bl      _start
        b       _start
        bx      lr
weakfn:
        bx      lr

--- a.s

        .type   _start, %function
_start:
        bl      thumbfn
        blx     thumbfn
        b       armfn
        bx      lr

        .text
        .thumb
        .globl  thumbfn
        .thumb_func
        .type   thumbfn, %function
thumbfn:
        bl      armfn
        bx      lr
        .arm
        .globl  armfn
        .type   armfn, %function
armfn:  bx      lr

=== literal pools loading another object's symbols
--- a.s
        .text
        .globl  _start
_start:
        ldr     r0, =target
        ldr     r1, =target + 8
        ldr     r2, =0x12345678
        ldr     r3, =weakfn
        adr     r4, _start
        bx      lr
        .ltorg
--- b.s
        .text
        .globl  target
        .weak   weakfn
target: bx      lr
weakfn: bx      lr

dptr:   .word   target
        .word   target + 16
        .word   target - .
        .word   wsym
        .word   com1
        .word   bssvar

        .size   dptr, . - dptr
after:  .word   0

        .text
        .globl  target

        .comm   com1, 8, 8
        .bss
        .globl  bssvar

        .size   bssvar, 16

=== the GOT and the position-independent idiom
--- a.s
        .text
        .globl  _start
        .type   _start, %function
_start:
        adr     r3, .LGOT
        ldr     r3, [r3]
.LPIC0: add     r3, pc, r3
        adr     r0, .Lgot
        ldr     r0, [r0]
        ldr     r0, [r3, r0]
        adr     r1, .Loff
        ldr     r1, [r1]
        add     r1, r3, r1
        adr     r2, .Lprel
        ldr     r2, [r2]
        bl      helper(PLT)
        b       helper(PLT)
        bx      lr
.LGOT:  .word   _GLOBAL_OFFSET_TABLE_ - (.LPIC0 + 8)
.Lgot:  .word   shared(GOT)
.Loff:  .word   shared(GOTOFF)
.Lprel: .word   shared(GOT_PREL)
        .word   helper(PLT)
--- b.s
        .text
        .globl  helper
        .type   helper, %function
helper: bx      lr
        .data
        .globl  shared
shared: .word   42

=== the halves of an address, absolute and against a label
--- a.s
        .text
        .globl  _start
        .type   _start, %function
_start:
        movw    r4, #:lower16:(_GLOBAL_OFFSET_TABLE_ - (.LPIC1 + 8))
        movt    r4, #:upper16:(_GLOBAL_OFFSET_TABLE_ - (.LPIC1 + 8))
.LPIC1: add     r4, r4, pc
        movw    r5, #:lower16:shared
        movt    r5, #:upper16:shared
        movw    r6, #:lower16:(shared - .LPIC2)
        movt    r6, #:upper16:(shared - .LPIC2)
.LPIC2: add     r6, r6, pc
        movw    r7, #:lower16:shared + 8
        movt    r7, #:upper16:shared + 8
        movw    r8, #:lower16:helper
        movt    r8, #:upper16:helper
        bx      lr
        .word   shared(GOT)
--- b.s
        .text
        .globl  helper
        .type   helper, %function
helper: bx      lr
        .data
        .globl  shared
shared: .word   42
        .word   43
        .word   44
