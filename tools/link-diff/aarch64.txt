=== calls and branches between objects
--- a.s
        .text
        .globl  _start
        .type   _start, %function
_start:
        bl      helper
        b       helper
        bl      weakfn
        b.eq    helper
        cbz     x0, helper
        tbz     x1, #3, helper
        bl      absent
        ret
        .weak   absent
--- b.s
        .text
        .globl  helper
        .weak   weakfn
        .type   helper, %function
helper:
        bl      _start
        b       _start
        b.ne    _start
        ret
weakfn:
        ret

=== page-relative addressing across objects
--- a.s
        .text
        .globl  _start
_start:
        adrp    x0, target
        add     x0, x0, :lo12:target
        adrp    x1, target + 16
        add     x1, x1, :lo12:(target + 16)
        adrp    x2, bssvar
        ldr     x2, [x2, :lo12:bssvar]
        adr     x3, _start
        adrp    x4, rodata
        add     x4, x4, :lo12:rodata
        ret
--- b.s
        .text
        .globl  target
target: nop
        nop
        nop
        nop
        .section .rodata
        .globl  rodata
rodata: .quad   0x1122334455667788
        .bss
        .globl  bssvar
bssvar: .space  16

=== data references between objects
--- a.s
        .data
        .globl  dptr
        .type   dptr, %object
dptr:   .quad   target
        .quad   target + 16
        .word   target
        .word   after - dptr
        .quad   wsym
        .quad   com1
        .quad   bssvar
        .size   dptr, . - dptr
after:  .quad   0
        .text
        .globl  _start
_start: ret
--- b.s
        .text
        .globl  target
        .type   target, %function
target: nop
        .weak   wsym
wsym:   nop
        .comm   com1, 8, 8
        .bss
        .globl  bssvar
        .type   bssvar, %object
bssvar: .space  16
        .size   bssvar, 16
