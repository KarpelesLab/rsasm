=== calls and branches between objects
--- a.s
        .syntax unified
        .text
        .thumb
        .globl  _start
        .thumb_func
        .type   _start, %function
_start:
        bl      helper
        b.w     helper
        beq.w   helper
        bne.w   weakfn
        bl      weakfn
        bl      absent
        bx      lr
        .weak   absent
--- b.s
        .syntax unified
        .text
        .thumb
        .globl  helper
        .weak   weakfn
        .thumb_func
        .type   helper, %function
helper:
        bl      _start
        b.w     _start
        bx      lr
        .thumb_func
weakfn:
        bx      lr

--- b.s
        .syntax unified
        .text
        .arm
        .globl  armfn

armfn:  bl      thumbfn
        bx      lr
        .thumb
        .globl  thumbfn
        .thumb_func
        .type   thumbfn, %function
thumbfn:
        bx      lr
        nop

=== literal pools loading another object's symbols

        .syntax unified
        .text
        .thumb

        .thumb_func
_start:
        ldr     r0, =target
        ldr     r1, =target + 8
        ldr     r2, =0x12345678
        ldr     r3, =weakfn

        bx      lr

        .syntax unified
        .text
        .thumb

target: bx      lr
        .thumb_func
weakfn: bx      lr

--- a.s
        .syntax unified
        .data
        .globl  dptr
        .type   dptr, %object
dptr:   .word   target
        .word   target + 16
        .word   target - .
        .word   wsym
        .word   com1
        .word   bssvar
        .word   after - dptr
        .size   dptr, . - dptr
after:  .word   0
        .syntax unified
        .text
        .thumb
        .globl  _start
        .thumb_func
_start: bx      lr
--- b.s
        .syntax unified
        .text
        .thumb
        .globl  target
        .thumb_func
        .type   target, %function
target: bx      lr
        .weak   wsym
        .thumb_func
wsym:   bx      lr
        .comm   com1, 8, 8
        .bss
        .globl  bssvar
        .type   bssvar, %object
bssvar: .space  16
        .size   bssvar, 16

=== a weak definition overridden by a strong one in another object
--- a.s
        .syntax unified
        .text
        .thumb
        .globl  _start
        .thumb_func
_start:
        bl      both
        bl      weakonly
        ldr     r0, =both
        bx      lr
        .ltorg
        .weak   both
        .thumb_func
both:   nop
        nop

        .text
        .thumb
        .globl  both
        .thumb_func
both:   bkpt    #1
        bkpt    #1
        .weak   weakonly
        .thumb_func
weakonly:
        bx      lr

=== the GOT and the position-independent idiom
--- a.s
        .syntax unified
        .thumb
        .text
        .globl  _start
        .thumb_func
        .type   _start, %function
_start:
        movw    r4, #:lower16:(_GLOBAL_OFFSET_TABLE_ - (.LPIC1 + 4))
        movt    r4, #:upper16:(_GLOBAL_OFFSET_TABLE_ - (.LPIC1 + 4))
.LPIC1: add     r4, pc
        movw    r5, #:lower16:shared
        movt    r5, #:upper16:shared
        movw    r6, #:lower16:(shared - .LPIC2)
        movt    r6, #:upper16:(shared - .LPIC2)
.LPIC2: add     r6, pc
        bl      helper(PLT)
        b.w     helper(PLT)
        bx      lr
        .p2align 2
        .word   _GLOBAL_OFFSET_TABLE_ - (.LPIC1 + 4)
        .word   shared(GOT)
        .word   shared(GOTOFF)
        .word   shared(GOT_PREL)
        .word   helper(PLT)
--- b.s
        .syntax unified
        .thumb
        .text
        .globl  helper
        .thumb_func
        .type   helper, %function
helper: bx      lr
        .data
        .globl  shared
shared: .word   42
