# SuperH programs for tools/link-diff.
#
# SuperH is the target whose relocations keep their addend in the field they
# cover -- BFD calls them `partial_inplace`, and its linker reads the field
# and ignores the entry's addend -- so an addend written into the wrong one
# of the two places is invisible until something links. These programs put an
# addend on every kind of reference: a call, a `mov.l` through the constant
# pool, and data.

=== calls and branches between objects
--- a.s
        .text
        .globl  _start
        .type   _start, @function
_start:
        bt      1f
        bf      1f
1:      mov.l   .Lhelper, r0
        jsr     @r0
        nop
        mov.l   .Lweak, r1
        jmp     @r1
        nop
        .align  2
.Lhelper:
        .long   helper
.Lweak: .long   weakfn
--- b.s
        .text
        .globl  helper
        .weak   weakfn
        .type   helper, @function
helper:
        rts
        nop
weakfn:
        rts
        nop

=== constant pool entries with addends on another object's symbols
--- a.s
        .text
        .globl  _start
_start:
        mov.l   .Ltarget, r0
        mov.l   .Ltarget8, r1
        mov.l   .Lbss, r2
        mov.l   .Lrodata, r3
        rts
        nop
        .align  2
.Ltarget:
        .long   target
.Ltarget8:
        .long   target + 8
.Lbss:  .long   bssvar + 4
.Lrodata:
        .long   rodata
--- b.s
        .text
        .globl  target
target: nop
        nop
        .section .rodata
        .globl  rodata
rodata: .long   0x11223344
        .bss
        .globl  bssvar
bssvar: .space  16

=== data references between objects
--- a.s
        .data
        .globl  dptr
        .type   dptr, @object
dptr:   .long   target
        .long   target + 16
        .long   target - 8
        .long   wsym
        .long   com1
        .long   bssvar
        .long   after - dptr
        .size   dptr, . - dptr
after:  .long   0
        .text
        .globl  _start
_start: rts
        nop
--- b.s
        .text
        .globl  target
        .type   target, @function
target: nop
        .weak   wsym
wsym:   nop
        .comm   com1, 8, 8
        .bss
        .globl  bssvar
        .type   bssvar, @object
bssvar: .space  16
        .size   bssvar, 16

=== a weak definition overridden by a strong one in another object
--- a.s
        .text
        .globl  _start
_start:
        bsr     both
        nop
        mov.l   .Lboth, r0
        jsr     @r0
        nop
        rts
        nop
        .align  2
.Lboth: .long   both
        .weak   both
both:   nop
        nop
--- b.s
        .text
        .globl  both
both:   sleep
        sleep
        .weak   weakonly
weakonly:
        rts
        nop

=== common symbols merged between objects
--- a.s
        .comm   shared, 4, 4
        .comm   bigger, 8, 8
        .text
        .globl  _start
_start:
        mov.l   .Lshared, r0
        mov.l   .Lbigger, r1
        rts
        nop
        .align  2
.Lshared:
        .long   shared
.Lbigger:
        .long   bigger
--- b.s
        .comm   shared, 4, 4
        .comm   bigger, 64, 32
        .text
        .globl  reader
reader:
        mov.l   .Lb, r0
        rts
        nop
        .align  2
.Lb:    .long   bigger + 32

=== refused: a branch to another object's symbol
--- a.s
        .text
        .globl  _start
_start:
        bsr     helper
        nop
        bra     helper
        nop
--- b.s
        .text
        .globl  helper
helper: rts
        nop

# A common block that names no alignment is aligned by its size, so the
# merged `grows` takes the larger object's 16; what `.lcomm` reserves is
# aligned by its size too, and a `.local` symbol's `.comm` as it asks.
=== common blocks aligned by their size, and objects reserved in .bss
--- a.s
        .comm   odd3, 3
        .comm   grows, 5
        .lcomm  lone1, 1
        .lcomm  lone6, 6
        .lcomm  lone9, 9
        .data
        .long   odd3, grows, lone1, lone6, lone9
        .text
        .globl  _start
_start:
        rts
        nop
--- b.s
        .comm   odd3, 3
        .comm   grows, 24
        .local  loc2
        .comm   loc2, 2, 8
        .lcomm  ltwo, 2
        .data
        .long   grows, loc2, ltwo
        .text
        .globl  reader
reader:
        rts
        nop
