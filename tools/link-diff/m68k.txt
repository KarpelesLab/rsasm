# The five thread-local access models, resolved by a static link. GNU ld
# gives the two dynamic ones a pair of GOT slots each and leaves the sequence
# as written, since nothing here calls through a PLT, and works out the
# offsets of the other three itself. `_start` reads, in order: general
# dynamic, with %a5 holding the GOT pointer and the call that reads the
# slots; local dynamic, one call for the module and then an offset per
# variable from the address it returns; initial exec, a GOT slot holding the
# offset from the thread pointer; and local exec, which the linker resolves
# outright. The two `.tdata` and the `.tbss` have to end up adjacent, which
# is why nothing else with contents follows them.
=== thread-local variables across objects
--- a.s
        .text
        .globl  _start
_start:
        lea     tv@TLSGD(%a5),%a0
        jsr     __tls_get_addr
        lea     tloc@TLSLDM(%a5),%a0
        jsr     __tls_get_addr
        lea     tloc@TLSLDO(%a0),%a1
        lea     tv2@TLSLDO(%a0),%a2
        move.l  tv2@TLSIE(%a5),%d0
        move.l  tb@TLSIE(%a5),%d1
        move.l  #tv3@TLSLE,%d2
        move.l  #tb@TLSLE,%d3
        lea     tv@TLSLE(%a5),%a3
        rts
        .section .tdata,"awT",@progbits
        .globl  tv
tv:     .long   0x11223344
tloc:   .long   99
--- b.s
        .text
        .globl  helper, __tls_get_addr
helper:
        move.l  #tv@TLSLE,%d0
        .long   tv@TLSLDO
        rts
__tls_get_addr:
        rts
        .section .tdata,"awT",@progbits
        .globl  tv2, tv3
tv2:    .long   2
tv3:    .long   3
        .section .tbss,"awT",@nobits
        .globl  tb
tb:     .space  16

# The position-independent suffixes, resolved by a static link. `%a5` holds
# the GOT pointer the psABI asks for, which the linker defines as
# `_GLOBAL_OFFSET_TABLE_`; `@GOT` then reads a slot of it, and `@GOTPC`
# reaches the same slot from the PC. `@PLTPC` on a branch resolves to the
# target, there being no dynamic linker to route it through a stub. Which
# slot of the GOT each symbol gets is the order the linker meets the
# relocations in, so the GOT pointer is loaded with an immediate: GNU as
# writes the fixup of a relaxable `lea` after every other one in the object,
# and a `lea` of the table would move both slots.
#
# `@PLT` is left out: GNU ld 2.47 fails a BFD assertion on `R_68K_PLTOFF32`
# in a static link, so there is no reference for it. So is a GOT slot for a
# local label, whose place among the global ones the m68k linker works out
# from the symbol's index within its object, which the section symbols GNU
# as writes and rsasm does not move; `tools/xas-diff/m68k-relocs.txt` has
# that relocation, where the two objects agree exactly.
=== position-independent references across objects
--- a.s
        .text
        .globl  _start
        .type   _start, @function
_start:
        move.l  #_GLOBAL_OFFSET_TABLE_,%a5
        move.l  gdata@GOT(%a5),%a0
        move.l  (%a0),%d0
        move.l  gdata@GOTPC(%pc),%d1
        move.l  other@GOT(%a5),%a1
        bsr.w   helper@PLTPC
        bra.w   helper@PLTPC
        rts
--- b.s
        .text
        .globl  helper
        .type   helper, @function
helper:
        move.l  gdata@GOT(%a5),%d0
        rts
        .data
        .globl  gdata, other
        .type   gdata, @object
gdata:  .long   0x11223344
other:  .long   0x55667788
