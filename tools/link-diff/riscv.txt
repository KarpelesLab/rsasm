=== calls and branches between objects
--- a.s
        .text
        .globl  _start
        .type   _start, @function
_start:
        call    helper
        call    weakfn
        jal     ra, helper
        j       helper
        beq     a0, a1, helper
        bne     a0, a1, weakfn
        tail    helper
        ret
--- b.s
        .text
        .globl  helper
        .weak   weakfn
        .type   helper, @function
helper:
        call    _start
        j       _start
        ret
weakfn:
        ret

=== pc-relative addressing of another object's symbols
--- a.s
        .text
        .globl  _start
_start:
        la      a0, target
        la      a1, bssvar
        lla     a2, target
        lui     a3, %hi(target)
        addi    a3, a3, %lo(target)
1:      auipc   a4, %pcrel_hi(target)
        addi    a4, a4, %pcrel_lo(1b)
        ret
--- b.s
        .text
        .globl  target
target: nop
        nop
        .bss
        .globl  bssvar
bssvar: .space  16

=== data references between objects
--- a.s
        .data
        .globl  dptr
        .type   dptr, @object
dptr:   .word   target
        .word   target + 16
        .word   wsym
        .word   com1
        .word   bssvar
        .word   after - dptr
        .size   dptr, . - dptr
after:  .word   0
        .text
        .globl  _start
_start: ret
--- b.s
        .text
        .globl  target
        .type   target, @function
target: nop
        .weak   wsym
wsym:   nop
        .comm   com1, 8, 8
        .bss
        .globl  bssvar
        .type   bssvar, @object
bssvar: .space  16
        .size   bssvar, 16

=== a weak definition overridden by a strong one in another object
--- a.s
        .text
        .globl  _start
_start:
        call    both
        call    weakonly
        la      a0, both
        ret
        .weak   both
both:   nop
        nop
--- b.s
        .text
        .globl  both
both:   ebreak
        ebreak
        .weak   weakonly
weakonly:
        ret

=== common symbols merged between objects
--- a.s
        .comm   shared, 4, 4
        .comm   bigger, 8, 8
        .text
        .globl  _start
_start:
        la      a0, shared
        la      a1, bigger
        ret
--- b.s
        .comm   shared, 4, 4
        .comm   bigger, 64, 32
        .text
        .globl  reader
reader:
        la      a0, bigger
        ret
