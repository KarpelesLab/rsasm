# MIPS programs for tools/link-diff.
#
# Every object here defines something in `.text`, `.data` and `.bss`. That is
# not decoration: llvm-mc creates all three in every object whether or not the
# source uses them, and on MIPS it gives them an alignment of 16, so an empty
# `.data` in the reference's second object pads the linked `.data` out by up
# to 15 bytes that rsasm -- which writes no empty sections, as
# `tools/mc-diff/canon.sh` records -- has nothing to match. Keeping each
# section non-empty on both sides leaves the comparison about relocations.

=== calls and branches between objects
--- a.s
        .text
        .set    noreorder
        .globl  _start
        .type   _start, @function
_start:
        jal     helper
        nop
        j       helper
        nop
        beq     $4, $5, helper
        nop
        bne     $4, $5, weakfn
        nop
        bal     helper
        nop
        jr      $31
        nop
--- b.s
        .text
        .set    noreorder
        .globl  helper
        .weak   weakfn
        .type   helper, @function
helper:
        jal     _start
        nop
        j       _start
        nop
        jr      $31
        nop
weakfn:
        jr      $31
        nop

=== halves of another object's address
--- a.s
        .text
        .set    noreorder
        .globl  _start
_start:
        lui     $4, %hi(target)
        addiu   $4, $4, %lo(target)
        lui     $5, %hi(bssvar)
        lw      $5, %lo(bssvar)($5)
        lui     $6, %hi(rodata)
        addiu   $6, $6, %lo(rodata)
        jr      $31
        nop
--- b.s
        .text
        .set    noreorder
        .globl  target
target: nop
        nop
        .section .rodata
        .globl  rodata
rodata: .word   0x11223344
        .bss
        .globl  bssvar
bssvar: .space  16

=== data references between objects
--- a.s
        .data
        .globl  dptr
        .type   dptr, @object
dptr:   .word   target
        .word   target + 16
        .word   wsym
        .word   com1
        .word   bssvar
        .word   after - dptr
        .size   dptr, . - dptr
after:  .word   0
        .text
        .set    noreorder
        .globl  _start
_start: jr      $31
        nop
--- b.s
        .text
        .set    noreorder
        .globl  target
        .type   target, @function
target: nop
        .weak   wsym
wsym:   nop
        .comm   com1, 8, 8
        .data
        .word   0x55667788
        .bss
        .globl  bssvar
        .type   bssvar, @object
bssvar: .space  16
        .size   bssvar, 16

=== a weak definition overridden by a strong one in another object
--- a.s
        .text
        .set    noreorder
        .globl  _start
_start:
        jal     both
        nop
        jal     weakonly
        nop
        lui     $4, %hi(both)
        addiu   $4, $4, %lo(both)
        jr      $31
        nop
        .weak   both
both:   nop
        nop
--- b.s
        .text
        .set    noreorder
        .globl  both
both:   break
        break
        .weak   weakonly
weakonly:
        jr      $31
        nop

# Thread-local storage. Only the two models that need no GOT are linked
# here: the link script this harness writes places no `.got` and defines no
# `_gp`, so GNU ld reports every `%tlsgd`, `%tlsldm` and `%gottprel` as
# "relocation truncated to fit" -- the offset is measured from a `_gp` that
# is zero. Whole objects for those three are compared in
# `tools/mc-diff/mips-relocs.txt` instead.
#
# `.tbss` is left out for the same kind of reason: this script lays the
# NOBITS sections out in the order the objects name them, which puts `.bss`
# between `.tdata` and `.tbss`, and the MIPS linker refuses a thread-local
# segment with a non-thread-local section inside it.
#
# No operand here carries an addend: llvm-mc, the reference for MIPS,
# drops the addend of a thread-local relocation, and under `REL` there is
# nowhere but the field for it, so its image would differ from one GNU as
# and rsasm both get right.
=== thread-local variables across objects
--- a.s
        .text
        .set    noreorder
        .globl  _start
_start:
        # Local exec, off the thread pointer `rdhwr` reads.
        rdhwr   $3, $29
        lui     $4, %tprel_hi(tv)
        addiu   $4, $4, %tprel_lo(tv)
        lui     $5, %tprel_hi(tv2)
        addiu   $5, $5, %tprel_lo(tv2)
        lw      $6, %tprel_lo(tloc)($3)
        # The offsets local dynamic adds to what `__tls_get_addr` returned,
        # which a static link resolves without the call.
        lui     $7, %dtprel_hi(tv2)
        addiu   $7, $7, %dtprel_lo(tv2)
        addiu   $8, $8, %dtprel_lo(tloc)
        jr      $31
        nop
        .data
        .word   0x11223344
        .bss
        .space  4
        .section .tdata,"awT",@progbits
        .globl  tv
tv:     .word   0x55667788
tloc:   .word   99
--- b.s
        .section .tdata,"awT",@progbits
        .globl  tv2
tv2:    .word   2
        .word   3
        .text
        .set    noreorder
        .globl  helper
helper:
        rdhwr   $2, $29
        lui     $4, %tprel_hi(tv)
        addiu   $4, $4, %tprel_lo(tv)
        lui     $5, %dtprel_hi(tv2)
        addiu   $5, $5, %dtprel_lo(tv2)
        jr      $31
        nop
        .data
        .word   0x99aabbcc
        .bss
        .space  8
